    # Download and run the official Homebrew installer
    /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"

echo ""

**** Zsh Configuration Files
These are documented in order they are loaded.

***** zshenv
Every shell (login, interactive, script). Loaded first, always.
The =~/.zshenv= is a symlink to =~/.config/zsh/.zshenv=, which bootstraps =ZDOTDIR= so zsh finds the rest of its config in =~/.config/zsh/=.
In other words, because we're using the XDG_CONFIG_HOME standard (~/.config directory), we need to symlink =.zshenv= which MUST be in the home directory.

Symlink:
#+begin_src fundamental :tangle "./symlink_dot_zshenv.tmpl"
{{ .chezmoi.homeDir }}/.config/zsh/.zshenv
#+end_src

#+begin_src shell :tangle "./dot_config/zsh/dot_zshenv"
export XDG_CONFIG_HOME=${XDG_CONFIG_HOME:=${HOME}/.config}
export ZDOTDIR=${ZDOTDIR:=${XDG_CONFIG_HOME}/zsh}
#+end_src

After a major mactex upgrade, TeX Live regenerates [[https://www.tug.org/texlive/doc/fmtutil.html][format files]] (=fmtutil=) and font map files (=updmap=) on first run. This blocks any process that calls =pdflatex= synchronously (including Emacs during org export). See [[https://tug.org/texlive/Contents/live/texmf-dist/doc/texlive/texlive-en/texlive-en.html#x1-320003.4.4][TeX Live Guide §3.4.4: System font configuration]] and [[https://tug.org/texlive/scripts-sys-user.html][updmap/fmtutil sys vs user]] for background. The script below warms the cache after each mactex upgrade.

#+begin_src shell :tangle "./.chezmoiscripts/darwin/run_onchange_before_04-warm-tex-cache.sh.tmpl" :shebang "#!/bin/bash"
# mactex version: {{ output "brew" "list" "--cask" "--versions" "mactex" | trim }}
# Warms TeX Live font and format caches on first run after install or upgrade.
# Without this, the first pdflatex call blocks synchronously while regenerating
# caches, which freezes any process waiting on it (e.g. Emacs during org export).
# Only warms pdflatex; to warm all engines (xelatex, lualatex, etc.) use instead:
#   fmtutil-user --all && updmap-user  (~90s)
set -e
pdflatex -interaction=nonstopmode /dev/null 2>/dev/null || true
#+end_src

#+begin_src shell :tangle "./dot_config/zsh/function.zsh"
# ecc: interactive emacsclient - reports server status, then opens a frame
ecs() {
  if [[ "$1" == "-k" ]]; then
    if ! emacsclient -e t &>/dev/null; then
      echo "no server running"
      return 1
    fi
    read "reply?kill server? [y/n] "
    if [[ "$reply" == "y" ]]; then
      emacsclient -e '(kill-emacs)'
      echo "server killed"
    else
      echo "aborted"
    fi

(after! evil
  (defun my/move-buffer-lines (count)
    "Move COUNT buffer lines, resolving hidden destinations without guessing."
    (let ((line-move-visual nil)
          (line-move-ignore-invisible nil))
      (evil-line-move count))
    (when (invisible-p (point))
      (if my/counted-motions-reveal
          (cond ((derived-mode-p 'org-mode) (org-fold-reveal))
                ((derived-mode-p 'markdown-mode) (markdown-show-entry))
                (t (+fold/open)))
        ;; Leave folds closed and stop at the boundary in the motion direction.
        (while (and (invisible-p (point))
                    (if (> count 0) (not (eobp)) (not (bobp))))
          (goto-char
           (if (> count 0)
               (next-char-property-change (point) (point-max))
             (max (point-min)
                  (1- (previous-char-property-change (point) (point-min)))))))
        ;; Org can leave a visible newline at the end of a hidden span.
        (when (and (> count 0) (eolp) (not (eobp)))
          (forward-char 1))))
    (evil-adjust-cursor))

(require 'org-roam-dailies)

(require 'seq)

(defvar my/org-weekly-do-directory "~/org/time/week/do/"
  "Directory for editable weekly Do drafts.")

(defun my/org-journal--daily-body (file)
  "Copy the body of daily FILE for a weekly draft without changing its source.
Include unsaved edits.  Remove file metadata and copied global IDs, resolve
relative file links, and nest the daily note's headings below its date."
  (with-temp-buffer
    (let ((source (find-buffer-visiting file)))
      (if source
          (insert (with-current-buffer source
                    (save-restriction
                      (widen)
                      (buffer-substring-no-properties (point-min) (point-max)))))
        (insert-file-contents file)))
    (delay-mode-hooks (org-mode))
    (goto-char (point-min))
    (when (looking-at-p org-property-drawer-re)
      (org-roam-end-of-meta-data)
      (delete-region (point-min) (point)))
    (dolist (keyword (reverse (org-element-map (org-element-parse-buffer)
                                 'keyword #'identity)))
      (when (and (not (org-element-lineage keyword '(headline)))
                 (member (org-element-property :key keyword)
                         '("TITLE" "DATE" "FILETAGS" "STARTUP" "PUBLISH" "DRAFT"
                           "ROAM_ALIASES" "ROAM_REFS")))
        (delete-region (org-element-property :begin keyword)
                       (org-element-property :end keyword))))
    (let ((org-odd-levels-only nil)
          (org-adapt-indentation nil))
      (org-map-entries (lambda () (org-entry-delete nil "ID")))
      ;; An unfinished, empty capture does not make a day nonempty.
      (dolist (heading (reverse (org-element-map (org-element-parse-buffer)
                                   'headline #'identity)))
        (when (and (string-empty-p (org-element-property :raw-value heading))
                   (not (org-element-property :todo-keyword heading))
                   (not (org-element-contents heading)))
          (delete-region (org-element-property :begin heading)
                         (org-element-property :end heading))))
      (org-map-entries #'org-demote))
    ;; Parse links so code and examples are left alone.  Edit from the end to
    ;; keep earlier element positions valid.
    (dolist (link (reverse (org-element-map (org-element-parse-buffer) 'link #'identity)))
      (let ((path (org-element-property :path link)))
        (when (and (equal (org-element-property :type link) "file")
                   (not (file-name-absolute-p path)))
          (let* ((begin (org-element-property :begin link))
                 (end (- (org-element-property :end link)
                         (org-element-property :post-blank link)))
                 (description-begin (org-element-property :contents-begin link))
                 (description (and description-begin
                                   (buffer-substring-no-properties
                                    description-begin
                                    (org-element-property :contents-end link))))
                 (application (org-element-property :application link))
                 (search (org-element-property :search-option link))
                 (target (concat "file" (and application (concat "+" application))
                                 ":" (expand-file-name path (file-name-directory file))
                                 (and search (concat "::" search)))))
            (delete-region begin end)
            (goto-char begin)
            (insert (org-link-make-string target description))))))
    (buffer-string)))

(defun my/org-journal--monday (day)
  "Return the Monday of the ISO week containing absolute calendar DAY."
  ;; Calendar represents Sunday as zero, so wrap it back to Monday.
  (- day (mod (1- (nth 1 (calendar-iso-from-absolute day))) 7)))

(defun my/org-journal--weeks (current-monday)
  "Return nonempty, completed Roam daily notes grouped into weekly bodies.
Use Roam's daily directory and date parser.  Exclude CURRENT-MONDAY and
later weeks.  Read visiting buffers without modifying or saving them."
  (let ((directory (expand-file-name org-roam-dailies-directory org-roam-directory))
        entries)
    (when (file-directory-p directory)
      (dolist (file (org-roam-dailies--list-files))
        (let ((date (org-roam-dailies-calendar--file-to-date file)))
          (when (and date (seq-every-p #'integerp date)
                     (calendar-date-is-valid-p date))
            (let ((day (calendar-absolute-from-gregorian date)))
              (when (< day current-monday)
                (let ((body (my/org-journal--daily-body file)))
                  (when (string-match-p "\\S-" body)
                    (push (cons day
                                (concat (format-time-string
                                         "* %Y-%m-%d %A\n"
                                         (encode-time 0 0 12 (nth 1 date) (car date)
                                                      (nth 2 date)))
                                        body "\n"))
                          entries)))))))))
    (setq entries (sort entries (lambda (a b) (< (car a) (car b)))))
    (mapcar (lambda (week)
              (cons (car week) (mapconcat #'cdr (cdr week) "\n")))
            (sort (seq-group-by (lambda (entry) (my/org-journal--monday (car entry)))
                                entries)
                  (lambda (a b) (< (car a) (car b)))))))

(defun my/org-journal--write-week (monday body)
  "Write a missing weekly draft for MONDAY from BODY and return its path.
Never replace an existing file or visiting buffer."
  (let* ((iso (calendar-iso-from-absolute monday))
         (week (format "%d-W%02d" (nth 2 iso) (car iso)))
         (destination (expand-file-name (concat week ".org")
                                        my/org-weekly-do-directory)))
    (unless (or (file-exists-p destination) (find-buffer-visiting destination))
      (let ((start (calendar-gregorian-from-absolute monday)))
        (with-temp-buffer
          (insert (format "#+TITLE: Week %d, %d\n#+DATE: <%04d-%02d-%02d>\n#+PUBLISH: time\n#+DRAFT: true\n\n"
                          (car iso) (nth 2 iso) (nth 2 start) (car start) (nth 1 start))
                  body)
          (unless (bolp) (insert "\n"))
          (make-directory (file-name-directory destination) t)
          ;; Refuse to overwrite even if another process creates it meanwhile.
          (write-region (point-min) (point-max) destination nil 'silent nil 'excl)))
      destination)))

(defun my/org-weekly-do-drafts (&optional today)
  "Create missing Do drafts for completed weeks with nonempty Roam dailies.
Copy whole daily notes into individual weekly Org files with DRAFT: true.
Skip the current Monday-Sunday week, future weeks, empty days, and existing
weekly files or visiting buffers.  Return new paths in chronological order.
When interactive, open the latest new draft.  TODAY is an optional date
string for calculating which weeks are complete; default to today."
  (interactive)
  (let* ((decoded (decode-time (if today (org-read-date nil t today) (current-time))))
         (current-monday
          (my/org-journal--monday
           (calendar-absolute-from-gregorian
            (list (nth 4 decoded) (nth 3 decoded) (nth 5 decoded)))))
         created)
    (dolist (week (my/org-journal--weeks current-monday))
      (let ((file (my/org-journal--write-week (car week) (cdr week))))
        (when file (push file created))))
    (setq created (nreverse created))
    (when (called-interactively-p 'interactive)
      (if created
          (progn
            (find-file (car (last created)))
            (message "Created %d weekly Do draft(s); opened the latest" (length created)))
        (message "No missing completed weeks with daily notes")))
    created))

;;; org-glossary-export-fix.el --- Preserve only authored glossary breaks -*- lexical-binding: t; -*-

(require 'org-glossary)

(defun org-glossary-export-fix--trim-paragraph (value _backend _info)
  "Remove a single structural ending newline from inline glossary VALUE.
Work on a copied paragraph, preserving authored internal and explicit
line breaks.  Multi-paragraph and other block definitions are unchanged."
  (when (and (consp value)
             (null (cdr value))
             (eq (org-element-type (car value)) 'paragraph))
    (let* ((paragraph (org-element-copy (car value) t))
           (last-child (last (org-element-contents paragraph))))
      (when (and (stringp (car last-child))
                 (string-suffix-p "\n" (car last-child)))
        (setcar last-child (substring (car last-child) 0 -1))
        (list paragraph)))))

(defun org-glossary-export-fix--export-term (original entry info)
  "Export glossary ENTRY with ORIGINAL using an isolated copy of INFO.
Normalize only inline glossary paragraphs with preserved line breaks,
after existing parse-tree filters.  Keep the original export cache."
  (if (not (plist-get info :preserve-breaks))
      (funcall original entry info)
    (let ((export-info (copy-sequence info)))
      (setq export-info
            (plist-put
             export-info :filter-parse-tree
             (append
              (remq #'org-glossary-export-fix--trim-paragraph
                    (plist-get info :filter-parse-tree))
              (list #'org-glossary-export-fix--trim-paragraph))))
      (funcall original entry export-info))))

;; org-glossary exports a single paragraph's children inline.  With \n:t,
;; its structural ending newline otherwise becomes an HTML/LaTeX break.
(unless (advice-member-p #'org-glossary-export-fix--export-term
                        'org-glossary--export-term)
  (advice-add 'org-glossary--export-term :around
              #'org-glossary-export-fix--export-term))

#+begin_src python :tangle "./dot_agents/hooks/executable_protect_literate_config.py" :shebang "#!/usr/bin/env python3"
"""Restore config.org when a Codex tool call changes human-authored prose."""

from __future__ import annotations

import hashlib
import json
import os
import re
import shutil
import stat
import subprocess
import sys
import tempfile
import time
import unittest
from pathlib import Path
from typing import Any

TARGET_PATH = "config.org"
SNAPSHOT_VERSION = 1
STALE_AFTER_SECONDS = 24 * 60 * 60
BEGIN_SOURCE = re.compile(br"[ \t]*#\+begin_src(?:[ \t].*)?", re.IGNORECASE)
END_SOURCE = re.compile(br"[ \t]*#\+end_src[ \t]*", re.IGNORECASE)


class GuardError(RuntimeError):
    """The protected state could not be snapshotted or restored safely."""


def main() -> int:
    if sys.argv[1:] == ["--self-test"]:
        suite = unittest.defaultTestLoader.loadTestsFromTestCase(GuardTests)
        return 0 if unittest.TextTestRunner(verbosity=2).run(suite).wasSuccessful() else 1

    try:
        payload = json.load(sys.stdin)
        cwd = Path(str(payload.get("cwd") or ".")).resolve()
        repo = find_repository(cwd)
        if repo is None:
            return 0

        tool_use_id = str(payload.get("tool_use_id") or "")
        if not tool_use_id:
            raise GuardError("hook input has no tool_use_id")

        root = snapshot_root(repo)
        root.mkdir(parents=True, exist_ok=True, mode=0o700)
        cleanup_stale_snapshots(root)
        snapshot = root / safe_id(tool_use_id)
        event = payload.get("hook_event_name")

        if event == "PreToolUse":
            create_snapshot(repo, snapshot, payload)
            return 0
        if event == "PostToolUse":
            message = validate_and_restore(repo, snapshot, tool_use_id)
            if message:
                print(message, file=sys.stderr)
                return 2
            return 0
        raise GuardError(f"unsupported hook event: {event!r}")
    except (GuardError, json.JSONDecodeError, OSError, subprocess.CalledProcessError) as error:
        print(f"Literate config reflector failed: {error}", file=sys.stderr)
        return 2


def find_repository(cwd: Path) -> Path | None:
    try:
        result = run_git(cwd, "rev-parse", "--show-toplevel")
    except subprocess.CalledProcessError:
        return None
    return Path(result.stdout.decode().strip()).resolve()


def run_git(
    repo: Path, *args: str, input: bytes | None = None
) -> subprocess.CompletedProcess[bytes]:
    return subprocess.run(
        ["git", "-C", str(repo), *args],
        input=input,
        check=True,
        capture_output=True,
    )


def snapshot_root(repo: Path) -> Path:
    result = run_git(repo, "rev-parse", "--git-path", "codex-hooks/protect-literate-config")
    path = Path(result.stdout.decode().strip())
    return path if path.is_absolute() else (repo / path).resolve()


def safe_id(value: str) -> str:
    return hashlib.sha256(value.encode()).hexdigest()


def protected_projection(content: bytes) -> bytes:
    """Return exact ordered bytes outside complete Org source blocks."""

    protected = bytearray()
    inside_source = False
    source_started_at = 0

    for line_number, line in enumerate(content.splitlines(keepends=True), 1):
        logical_line = line.rstrip(b"\r\n")
        if inside_source:
            if END_SOURCE.fullmatch(logical_line):
                inside_source = False
            continue
        if END_SOURCE.fullmatch(logical_line):
            raise GuardError(f"unmatched #+end_src at line {line_number}")
        if BEGIN_SOURCE.fullmatch(logical_line):
            inside_source = True
            source_started_at = line_number
            continue
        protected.extend(line)

    if inside_source:
        raise GuardError(f"unclosed #+begin_src at line {source_started_at}")
    return bytes(protected)


def cleanup_stale_snapshots(root: Path) -> None:
    cutoff = time.time() - STALE_AFTER_SECONDS
    for child in root.iterdir():
        try:
            if child.is_dir() and child.stat().st_mtime < cutoff:
                shutil.rmtree(child)
        except OSError:
            pass


def create_snapshot(repo: Path, destination: Path, payload: dict[str, Any]) -> None:
    target = repo / TARGET_PATH
    info = target.lstat()
    if not stat.S_ISREG(info.st_mode):
        raise GuardError(f"{TARGET_PATH} is not a regular file")

    content = target.read_bytes()
    projection = protected_projection(content)
    if destination.exists():
        shutil.rmtree(destination)
    temporary = destination.with_name(f".{destination.name}.{os.getpid()}.tmp")
    if temporary.exists():
        shutil.rmtree(temporary)
    temporary.mkdir(parents=True, mode=0o700)

    metadata = {
        "version": SNAPSHOT_VERSION,
        "created_at": time.time(),
        "repo": str(repo),
        "tool_use_id": payload.get("tool_use_id"),
        "session_id": payload.get("session_id"),
        "turn_id": payload.get("turn_id"),
        "mode": stat.S_IMODE(info.st_mode),
        "content_sha256": digest(content),
        "projection_sha256": digest(projection),
    }
    write_private(temporary / "config.org", content)
    write_private(
        temporary / "manifest.json",
        json.dumps(metadata, sort_keys=True, separators=(",", ":")).encode(),
    )
    write_private(temporary / "index.entries", protected_index(repo))
    write_private(temporary / "integrity.sha256", snapshot_integrity(temporary).encode())
    os.replace(temporary, destination)


def validate_and_restore(repo: Path, snapshot: Path, tool_use_id: str) -> str | None:
    if not snapshot.is_dir():
        raise GuardError("matching pre-tool snapshot is missing")

    recorded_integrity = (snapshot / "integrity.sha256").read_text()
    if snapshot_integrity(snapshot) != recorded_integrity:
        raise GuardError("snapshot content or metadata was modified")

    metadata = json.loads((snapshot / "manifest.json").read_text())
    if (
        metadata.get("version") != SNAPSHOT_VERSION
        or metadata.get("repo") != str(repo)
        or metadata.get("tool_use_id") != tool_use_id
    ):
        raise GuardError("snapshot metadata does not match this tool call")

    before = (snapshot / "config.org").read_bytes()
    if digest(before) != metadata.get("content_sha256"):
        raise GuardError("snapshotted config.org content was modified")
    before_projection = protected_projection(before)
    if digest(before_projection) != metadata.get("projection_sha256"):
        raise GuardError("snapshotted prose projection was modified")

    target = repo / TARGET_PATH
    after_projection: bytes | None = None
    try:
        if target.is_file() and not target.is_symlink():
            after_projection = protected_projection(target.read_bytes())
    except GuardError:
        after_projection = None

    if after_projection == before_projection:
        shutil.rmtree(snapshot)
        return None

    restore_file(target, before, int(metadata["mode"]))
    restore_index(repo, (snapshot / "index.entries").read_bytes())
    shutil.rmtree(snapshot)
    return (
        "Restored config.org because this model tool call changed human-authored "
        "content outside Org source blocks. Retry with a source-block-only edit."
    )


def digest(content: bytes) -> str:
    return hashlib.sha256(content).hexdigest()


def write_private(path: Path, content: bytes) -> None:
    descriptor = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
    with os.fdopen(descriptor, "wb") as stream:
        stream.write(content)


def snapshot_integrity(snapshot: Path) -> str:
    checksum = hashlib.sha256()
    for name in ("config.org", "manifest.json", "index.entries"):
        checksum.update(name.encode())
        checksum.update(b"\0")
        checksum.update((snapshot / name).read_bytes())
        checksum.update(b"\0")
    return checksum.hexdigest()


def protected_index(repo: Path) -> bytes:
    return run_git(repo, "ls-files", "--stage", "-z", "--", TARGET_PATH).stdout


def restore_file(target: Path, content: bytes, mode: int) -> None:
    target.parent.mkdir(parents=True, exist_ok=True)
    descriptor, temporary_name = tempfile.mkstemp(prefix=f".{target.name}.", dir=target.parent)
    temporary = Path(temporary_name)
    try:
        with os.fdopen(descriptor, "wb") as stream:
            stream.write(content)
            stream.flush()
            os.fsync(stream.fileno())
        os.chmod(temporary, mode)
        os.replace(temporary, target)
    finally:
        if temporary.exists():
            temporary.unlink()


def restore_index(repo: Path, entries: bytes) -> None:
    run_git(repo, "update-index", "--force-remove", "--", TARGET_PATH)
    if entries:
        run_git(repo, "update-index", "-z", "--index-info", input=entries)

# Configure Dotfiles

## Model

Run `chezmoi source-path` to locate the source repository. Use that directory as
the working directory for the entire task.

`config.org` is the source of truth. Files such as `dot_config/**`, `Brewfile`, and
`.chezmoiscripts/**` are tangled outputs. Keep prose and code together in the Org
document. Never edit a tangled output directly.

## Workflow

1. Inspect `git status --short`, the relevant Org section, and its current tangle
   target. Preserve unrelated user changes.
2. Edit only Org source blocks in `config.org`, including the exact `:tangle`
   target. New blocks may be added under an existing human-authored section. Ask
   the user to make any required prose or heading changes. Add a shebang when the
   target must be executable.
3. Run `~/.agents/skills/configure-dotfiles/scripts/tangle.sh`. It uses a running
   Emacs server when available and otherwise launches batch Emacs. Do not
   reproduce the underlying Emacs invocation manually.
4. Run `git diff --check`, `git status --short`, and `git diff`. Confirm every
   generated change follows from the requested Org edits.
5. Preview deployment with `chezmoi apply --dry-run --verbose --force`. On normal
   command-line workflows, use `--force` only with `--dry-run`; it reveals target
   drift without overwriting it.
6. If implementation was requested and the preview contains only intended
   changes, apply the narrowest affected target or directory with `chezmoi apply
   --verbose <target>`. The interactive `M-x tangle-moi` helper may use `--force`
   for its real apply only after showing the full forced dry-run and receiving
   explicit confirmation. This avoids a second prompt that requires a TTY.
7. Verify installed targets and report source changes, applied targets, and
   anything intentionally left unapplied.

`M-x tangle-moi` is the interactive interface to the same tangle core installed
at `~/.config/doom/lisp/tangle-moi.el`.

## Stop conditions

Stop if the tangle script reports unsaved changes in an Emacs buffer visiting
`config.org`; preserve that buffer and ask the user to resolve it. Ask before
applying when the preview would overwrite independently modified home files,
delete targets, change permissions unexpectedly, execute package or setup
scripts, or include changes outside the request. Do not apply when the user asked
only for review or diagnosis.

## Conventions

- Use chezmoi naming conventions for destination paths (`dot_`, `private_`,
  `symlink_`, and `.tmpl`).
- Keep platform-specific scripts under `.chezmoiscripts/<os>/` and guard them
  appropriately.
- Keep secrets and host-specific credentials out of the repository.
- Prefer idempotent scripts with `set -e` and explicit paths.
- Flag nearby prose that needs a human update; do not edit it.

## Validation checklist

- The Org source parses and tangles successfully.
- The generated diff contains no unrelated changes.
- Templates and scripts remain executable where intended.
- The chezmoi dry-run reports only expected target changes.

description: Catalog PDF and EPUB sources in ~/org/sources with verified bibliography metadata, preserving abstracts and existing references.

---

# citation-add

Adds uncataloged PDF and EPUB files recursively from `~/org/sources/` to
`~/org/bib/references.bib`. Process one source completely before moving to the next.

The source root is an inbox. Place cataloged books, edited collections, and
book chapters in `books/`; journal/conference papers and research manuscripts
in `papers/`; theses/dissertations in `theses/`; technical/industry reports in
`reports/`; saved web pages in `web/`; and complete magazine/journal issues in
`periodicals/`. The root is for incoming or unresolved files.

Use accurate bibliography types. For generic `@misc` entries, set a verified
`entrysubtype` such as `preprint`, `report`, or `magazine` when the source's role
is known. A preprint remains a preprint even though it shares `papers/` with
published articles. `source_library.py` contains the shared type/subtype map.
PDF/EPUB extensions do not determine classification.

Retained scans in `originals/` are alternate attachments of the same work, not
new bibliography entries. List both paths in that entry's semicolon-separated
`file` field and preserve the organized reading copy as the primary attachment.

Run `python3 ~/.agents/skills/citation-add/audit_sources.py` to check catalog
coverage, attachment paths, and placement. Inbox items are reported separately
from broken paths or misplaced sources.
Entry quality is baked in at add time - no separate verify pass needed.

---

## Step 1: Clean and inventory

Strip Anna's Archive suffixes:

```bash
python3 ~/.agents/skills/citation-add/clean_filenames.py --apply
```

Find uncataloged PDFs and EPUBs:

```bash
python3 ~/.agents/skills/citation-add/find_missing.py
```

- `MISSING:<csv>` - source-root-relative paths to process (stop here if empty)
- `CITEKEYS:<csv>` - existing citekeys (keep for collision checks)

---

## Step 2: For each missing source - one at a time

### 2a. Classify and extract

For an EPUB, inspect the actual package metadata and spine text:

```bash
python3 ~/.agents/skills/citation-add/inspect_epub.py ~/org/sources/<file.epub> --text
```

Use the title, creators, publisher, date, and identifiers as evidence, then
verify the correct entry type and edition. A publisher description is not an
abstract. Never send EPUBs to `pdftotext`, `pdfinfo`, PDF OCR, or bbox helpers.
Use its metadata and text for the fallback workflow, omitting `abstract` unless
the work explicitly supplies one. Org Noter EPUB annotation requires nov.el;
cataloging and source inspection do not require that reader.

For a PDF, keep the existing extraction workflow:

```bash
python3 ~/.agents/skills/citation-add/classify_pdf.py \
  --pdf ~/org/sources/<filename> \
  --task extract:bib
```

Parse output:
- `CLASS:<class>` - suggested document class; treat as a hint only. Use content
  and venue to determine the correct BibTeX entry type (`@article`, `@inproceedings`,
  `@incollection`, `@book`, `@phdthesis`, etc.) - CLASS can be wrong. If it is,
  fix the rule in `classify_pdf.py` and note the case in `classify.md`.
- `DOI:<doi>` - go to Step 3a if present
- `META:title=...` / `META:author=...` - PDF metadata; use it when it is a real
  value (it is sometimes a file name like `main-all.dvi`)
- `FIELDS:<csv>` - fields still needed from page content
- `RANGE:<range>` - pages to read for fallback
- `NOTEXT:...` - scanned/encrypted; do NOT skip. Instead:
  1. Check the filename for embedded metadata (Anna's Archive names include author, title, year, publisher, ISBN)
  2. Use world knowledge for well-known works (classic books, famous papers)
  3. Only escalate to the user if neither source yields enough to build a complete entry

### 2b. Duplicate check

Once DOI or title is known, run:

```bash
python3 ~/.agents/skills/citation-add/check_duplicate.py \
  --bib ~/org/bib/references.bib \
  --doi "<doi-or-empty>" \
  --title "<title-or-empty>" \
  --author "<author-or-empty>" \
  --year "<year-or-empty>" \
  --new-file "<filename>"
```

Interpret the output token:

- `NO_MATCH` - no duplicate found, proceed to Step 3
- `ADD_FORMAT` - the DOI identifies the same work in another format. Preserve
  both files and associate both paths with the existing entry using a
  semicolon-separated `file` field. Do not create a duplicate entry or delete
  either format. For inferred matches across PDF/EPUB formats, first verify
  the work and edition, then follow this same multiple-attachment policy.
- `KEEP_EXISTING: citekey=... existing_file=... reason=...`
  - Delete the new PDF silently, log the decision in the Step 8 report
  - Do NOT add a new bib entry
- `KEEP_NEW: citekey=... existing_file=... reason=...`
  - Update the existing bib entry's `file` field to point to the new PDF
  - Delete the old PDF file, log the decision
  - Do NOT add a new bib entry
- `INFER: citekey=... existing_file=... new_pdf=... hint=...`
  - Read the opening pages or EPUB sections of both sources
  - Reason about whether they are the same work, using the `hint` as guidance
  - Resolve as KEEP_EXISTING, KEEP_NEW, or escalate to ASK if still unclear
  - If resolved, log the decision; if escalating, show the user the evidence
- `ASK: citekey=... existing_file=... message=...`
  - Show the user the message and wait for their choice before proceeding

,**Never silently overwrite a `file` field** without going through this flow.

---

## Step 3a: DOI path (preferred)

```bash
python3 ~/.agents/skills/citation-add/fetch_doi.py <doi>
```

Parse `FIELD:key=value` lines. Also watch for:
- `SPARSE:...` - CrossRef returned fewer than 4 substantive fields; read PDF pages
  to fill gaps before building the entry.

Then:

,**CrossRef year is authoritative.** Use it for both the `year` field and the
citekey year. If CrossRef returns `published-print` and `published-online` in
different years, use `published-print`. If only `published-online` is present,
use that.

Check citekey year matches CrossRef year before writing. If they differ, use
CrossRef year in both the citekey and the `year` field.

Format following `~/.agents/skills/citation-add/examples.md`:
- `journal` for articles (not `journaltitle`)
- `publisher` in `note = {Publisher: X}` for `@article`
- `publisher = {IEEE}` / `publisher = {ACM}` valid for `@inproceedings`
- `abstract`: follow **Abstract** below

Add `file = {~/org/sources/<filename>}` (temporary - will be updated after rename).

---

## Step 3b: Fallback path (no DOI)

1. Read the `RANGE:` pages from the PDF
2. Use `META:` fields as starting point; fill `FIELDS:` from page content
3. Get the `abstract` as described in **Abstract** below, never by inference
4. Read `~/.agents/skills/citation-add/examples.md` for formatting conventions
5. Build entry manually

,**Citekey convention:** `authoryear-slug`
- `year` = publication year (not submission/acceptance year)
- `slug` = 1-4 hyphenated words that answer "what would I search for?"
- Drop filler words: "towards", "combining", "using", "a", "an", "the", "on"
- Example: "Towards Responsible Software Engineering: combining value-based
  processes..." → `kapferer2024responsible-se`
- Check `CITEKEYS` for collisions; append `b`, `c` if needed

---

## Abstract (both paths)

The site shows `abstract` under an "Abstract" label as the paper's own text, so
it is either the work's own abstract, verbatim and complete, or omitted. Never
write it yourself: no paraphrase, summary, condensing, or truncation, and never
end it with "...". Summaries go in `annote` (Step 5).

Take the first source that has it:

1. CrossRef: the `FIELD:abstract=` line from `fetch_doi.py`.
2. arXiv, for `10.48550/arXiv.*` DOIs: the `<summary>` of
   `https://export.arxiv.org/api/query?id_list=<arxiv-id>`.
3. The PDF:

   ```bash
   python3 ~/.agents/skills/citation-add/extract_abstract.py ~/org/sources/<filename>
   ```

   - `ABSTRACT:<text>` - use it, after the check below
   - `NOABSTRACT:<reason>` - omit the field
   - `FAIL:<reason>` - get the pages to read from
     `classify_pdf.py --pdf <pdf> --task extract:abstract` (`patterns.md` notes
     where abstracts sit by document type) and copy the abstract exactly as
     printed. If a new pattern would have caught it, add the pattern to
     `extract_abstract.py` and record it in `patterns.md`.

Before writing it, check the text against the PDF page:

- It is the whole abstract. Nothing between it and the next heading
  (Keywords, Index Terms, CCS Concepts, 1 Introduction) is missing.
- No text-layer damage: dropped ligatures (`dene` for `define`), run-together
  or split words, lost line-end hyphens, OCR errors. Fix them to match the page.
- No publisher boilerplate: drop trailing copyright lines
  (`© 2015 Elsevier B.V. All rights reserved.`), received/accepted dates,
  funding or supplemental-material notes, and keyword lists.

Omit `abstract` when the work has none: books, theses whose only summary runs
over several pages, and columns, essays or magazine articles without one. For a
magazine article with no printed abstract, the publisher's registered abstract
(CrossRef, IEEE Xplore) may be used if it reads coherently; otherwise omit it.

---

## Step 4: Write entry and organize the source

Append the completed entry to the bib.

Keep the original file format and name it `{citekey}.pdf` or `{citekey}.epub`.
Use the shared type map to choose its folder; unmapped types stay in the inbox.

```python
from pathlib import Path
import sys
sys.path.insert(0, str(Path.home() / '.agents/skills/citation-add'))
from source_library import SOURCE_ROOT, category_for
source = Path('<original-full-path>').expanduser()
category = category_for('<bib-entry-type>', '<entrysubtype-or-empty>')
directory = SOURCE_ROOT / category if category else SOURCE_ROOT
directory.mkdir(parents=True, exist_ok=True)
destination = directory / ('<citekey>' + source.suffix.lower())
if destination != source:
    if destination.exists():
        raise FileExistsError(destination)
    source.rename(destination)
print(destination)
```

Update only the attachment being added in the entry's `file` field to this
actual destination. Preserve other attachments. Run the source audit again.

---

## Step 5: Annote

Read the abstract (from the entry, or the PDF's opening pages when the work has none).
Write a 2-3 sentence `annote`: plain-language summary written for your own future
reference. The `annote` is the only place for a summary; the site shows it as
"Brief", separate from the verbatim "Abstract".

A good annote answers: **why would I cite this?** It names the specific finding,
claim, or contribution - not just the topic. Prefer concrete over vague:

- Bad: "Discusses the role of automation in decision-making."
- Good: "Shows that automation bias persists even in expert users and cannot be
  eliminated by training - only by redesigning the automation itself."

If the paper is a foundational reference (Simon 1956, Bainbridge 1983), note what
it established and why it remains cited. If it's empirical, name the key result.
If it's a framework or model, say what it lets you do.

Append `annote = {<text>}` to the entry before validating.

---

## Step 6: Audit the abstract

Skip this step if the entry has no `abstract`. Otherwise check it against the
work's own text:

```bash
python3 ~/.agents/skills/citation-add/audit_abstracts.py fetch ~/org/bib/references.bib --keys <citekey>
python3 ~/.agents/skills/citation-add/audit_abstracts.py score ~/org/bib/references.bib --keys <citekey>
```

Act on the result line for the citekey:

- `VERBATIM` - done.
- `REPLACE` - the abstract is not the work's text or is incomplete, and the
  line names a verified source. Write that source's text, then compare it with
  the page: text taken from a PDF (`-> pdf_cut`) can still carry text-layer
  damage such as a mis-scanned quote, so apply the checks under **Abstract**:

  ```bash
  python3 ~/.agents/skills/citation-add/audit_abstracts.py apply ~/org/bib/references.bib --keys <citekey>
  ```

- `DELETE` - the script found no abstract for the work, and the bib text is not
  in the PDF. Check the first page; if the work really has none, run `apply` as
  above to remove the field.
- `REVIEW` - the script cannot settle it: a scanned or column-scrambled PDF, an
  abstract with no heading after it, or sources that disagree. Run
  `audit_abstracts.py show ~/org/bib/references.bib <citekey>` to see the bib
  text next to every source. Compare with the PDF page itself (if the text
  layer is damaged, render it with `pdftoppm -r 110 -f 1 -l 1 -png <pdf> <dir>/page`
  and read the image). Make the bib text match the page exactly, or remove the
  field, then record the check:

  ```bash
  python3 ~/.agents/skills/citation-add/audit_abstracts.py reviewed ~/org/bib/references.bib <citekey> "<what you checked>"
  ```

The audit is a guard, not an oracle. It reliably catches paraphrases, condensed
copies and missing sentences, and it says REVIEW rather than guessing when its
sources are weak. It sends no personal data: a generic User-Agent and no email.

---

## Step 7: Validate after each entry

Run biber after every single entry is added - catch errors immediately, not in bulk:

```bash
biber --tool --validate-datamodel ~/org/bib/references.bib 2>&1 | grep -E "WARN|ERROR"
rm -f ~/org/bib/references_bibertool.bib ~/org/bib/references.bib.blg
```

Fix all ERRORs before moving to the next PDF.
Fix WARNs if possible; note if not.

Target: 0 errors, 0 warnings.

---

## Step 8: Report

After all PDFs are processed:

- Each added citekey: DOI path vs fallback, CrossRef year correction if any
- Abstract source per entry (CrossRef, arXiv, extractor, page read) or why it was omitted
- Audit result per entry, and what was done for REPLACE, DELETE and REVIEW
- Duplicates found and how resolved
- PDFs skipped (NOTEXT, confirmed duplicates)
- Final biber validation status

---

## Re-auditing the whole bib

To check every existing abstract (for example after editing entries by hand),
run `fetch` and `score` without `--keys`. The first full fetch takes a few
minutes; later runs reuse the cache in `~/.cache/citation-add/abstract-audit/`.
`score` prints every entry that is not VERBATIM and writes `report.tsv` there.

- Settle REVIEW rows as in Step 6 and record each with `reviewed`.
- `apply` without `--keys` writes all REPLACE and DELETE results. It refuses to
  run while REVIEW rows remain unless they are covered by an overrides file.
- When the right text needs a source other than the one the script picked,
  put it in an overrides JSON (format in the script's `source_text` docstring:
  `from` is `crossref`, `openalex`, `arxiv`, `cut:N` or `span` with `start`/`end`
  anchors, plus optional `edits` and `drop_from`). Check each with
  `preview <file> <citekey>`, then run `apply --overrides <file>`.

Validate with biber afterwards (Step 7).

# Bibtex Entry Examples

Real cases from this bib. When generating a new entry, find the matching case
and follow it exactly - field names, field order, formatting conventions.

Abstracts are shown as a placeholder to keep the examples short. In a real
entry the `abstract` is the work's own abstract, copied verbatim and complete
(see "Abstract" in SKILL.md), or the field is omitted. Never shorten, reword or
summarize it: the site shows it under an "Abstract" label as the paper's own
text. Summaries belong in `annote`, which the site shows as "Brief".

---

## 1. arXiv preprint

,**Situation**: Paper only on arXiv. Filename like `2602.14270v1.pdf` or
`descriptive-name-2602.14270v1.pdf`. DOI is always `10.48550/arXiv.XXXX.XXXXX`.

,**Rule**: Use `@article` (our convention). Set `journal` to
`arXiv preprint arXiv:XXXX.XXXXX`. Always include the arXiv DOI.

```bibtex
@article{batista2026sycophantic-ai,
  author    = {Batista, Rafael M. and Griffiths, Thomas L.},
  title     = {A Rational Analysis of the Effects of Sycophantic AI},
  journal   = {arXiv preprint arXiv:2602.14270},
  year      = {2026},
  keywords  = {sycophancy, llm, belief-updating, confirmation-bias, epistemic-risk},
  abstract  = {<the work's own abstract, copied verbatim and complete>},
  file      = {~/org/sources/papers/sycophantic-ai-effects-2602.14270v1.pdf},
  doi       = {10.48550/arXiv.2602.14270}
}
```

---

## 2. Journal article (named publisher)

,**Situation**: Paper in a named journal published by Elsevier, Nature, Springer,
ACM, etc. The `publisher` field is invalid on `@article` per biber's datamodel.

,**Rule**: Use `@article`. Put the publisher in `note = {Publisher: X}` to
preserve the information without triggering a validation warning.

```bibtex
@article{babaei2025nasa-tlx,
  author    = {Babaei, Ebrahim and Dingler, Tilman and Tag, Benjamin and Velloso, Eduardo},
  title     = {Should we use the NASA-TLX in HCI? A review of theoretical and
               methodological issues around Mental Workload Measurement},
  journal   = {International Journal of Human-Computer Studies},
  volume    = {201},
  pages     = {103515},
  year      = {2025},
  note      = {Publisher: Elsevier},
  keywords  = {mental-workload, nasa-tlx, hci, cognitive-load, measurement},
  abstract  = {<the work's own abstract, copied verbatim and complete>},
  file      = {~/org/sources/papers/1-s2.0-S1071581925000722-main.pdf},
  doi       = {10.1016/j.ijhcs.2025.103515}
}
```

Filename heuristics: `1-s2.0-S...` = Elsevier ScienceDirect,
`s41597-...` = Nature/Springer.

---

## 3. IEEE conference paper

,**Situation**: Paper in IEEE proceedings (ICSA, ICSE, ICSME, etc.). DOI
typically in the footer of page 1. Filename often descriptive.

,**Rule**: Use `@inproceedings`. `publisher = {IEEE}` is valid here. Include
full conference name in `booktitle`.

```bibtex
@inproceedings{ivers2025genai-architecture,
  author    = {Ivers, James and Ozkaya, Ipek},
  title     = {Will Generative AI Fill the Automation Gap in Software Architecting?},
  booktitle = {2025 IEEE 22nd International Conference on Software Architecture
               Companion (ICSA-C)},
  year      = {2025},
  pages     = {41--48},
  publisher = {IEEE},
  keywords  = {software-architecture, genai, llm, automation, architecture-decisions},
  abstract  = {<the work's own abstract, copied verbatim and complete>},
  file      = {~/org/sources/papers/Will_Generative_AI_Fill_the_Automation_Gap_in_Software_Architecting.pdf},
  doi       = {10.1109/ICSA-C65153.2025.00014}
}
```

## 4. ACM conference paper

,**Situation**: Paper in ACM proceedings (FAccT, CHI, PLDI, etc.). Filename may
be an ACM DL identifier like `3531146.3533088.pdf`.

,**Rule**: Use `@inproceedings`. `publisher = {ACM}` is valid. Use full
proceedings title in `booktitle`.

```bibtex
@inproceedings{weidinger2022taxonomy,
  author    = {Weidinger, Laura and Uesato, Jonathan and Rauh, Maribeth and
               Griffin, Conor and Huang, Po-Sen and Mellor, John and ...},

  year      = {2022},
  publisher = {ACM},
  keywords  = {language-models, responsible-ai, risk-taxonomy, safety, ethics},
  abstract  = {<the work's own abstract, copied verbatim and complete>},
  file      = {~/org/sources/papers/3531146.3533088.pdf},
  doi       = {10.1145/3531146.3533088}
}
```

## 5. OpenReview / ICLR / NeurIPS workshop paper

,**Situation**: Paper at ICLR, NeurIPS, ICML, or similar ML venue that
publishes via OpenReview. No traditional publisher. DOI may be an arXiv DOI
if only an arXiv version is available.

,**Rule**: Use `@inproceedings`. Omit `publisher` entirely - these venues have
no publisher in the traditional sense. Use the conference short name (e.g.
`ICLR 2025`) in `booktitle` if the full proceedings title is not on the PDF.

```bibtex
@inproceedings{sharma2024sycophancy,
  author    = {Sharma, Mrinank and Tong, Meg and Korbak, Tomasz and ...},
  title     = {Towards Understanding Sycophancy in Language Models},
  booktitle = {ICLR 2024},
  year      = {2024},
  keywords  = {sycophancy, llm, rlhf, human-feedback, alignment},
  abstract  = {<the work's own abstract, copied verbatim and complete>},
  file      = {~/org/sources/papers/2310.13548v4.pdf},
  doi       = {10.48550/arXiv.2310.13548}
}
```

---

## 6. SSRN preprint

,**Situation**: Paper on SSRN. Filename like `ssrn-XXXXXXX.pdf`. Version
string like `v20260111` may appear on the title page. No DOI on first page.

,**Rule**: Use `@misc`. No `publisher`. Omit `doi` if not found on any page
read - do not guess it.

```bibtex
@misc{shaw2026cognitive-surrender,
  author    = {Shaw, Steven D. and Nave, Gideon},
  title     = {Thinking---Fast, Slow, and Artificial: How AI is Reshaping
               Human Reasoning and the Rise of Cognitive Surrender},
  year      = {2026},
  keywords  = {cognitive-surrender, dual-process, human-ai-interaction,
               reasoning, tri-system-theory},
  abstract  = {<the work's own abstract, copied verbatim and complete>},
  file      = {~/org/sources/ssrn-6097646.pdf}
}
```

---

## 7. Never invent a DOI

,**Situation**: DOI field not visible on page 1 or last page.

,**Rule**: Omit the `doi` field entirely. A missing DOI is harmless. A
hallucinated DOI (e.g. `10.1109/XXX.0000.0000000`) silently breaks citation
linking and is worse than no DOI.

Wrong:
```bibtex
  doi = {10.1109/XXX.0000.0000000}   % placeholder - DO NOT DO THIS
  doi = {10.1145/unknown}             % guessed - DO NOT DO THIS
```

Correct: omit the field if unsure.

---

## 8. Book (commercial, no abstract)

,**Situation**: A commercially published book (Addison-Wesley, O'Reilly, MIT Press, etc.).
No abstract field - books don't have abstracts. ISBN goes in `note`.

,**Rule**: Use `@book`. Include `publisher`, `year`, `keywords`, `note` (for ISBN), `file`.
Omit `abstract` entirely. Omit `doi` unless a DOI is printed in the book itself.

```bibtex
@book{humble2010continuous-delivery,
  author    = {Humble, Jez and Farley, David},
  title     = {Continuous Delivery: Reliable Software Releases Through Build, Test, and Deployment Automation},
  publisher = {Addison-Wesley},
  year      = {2010},
  keywords  = {continuous-delivery, continuous-integration, deployment, build-automation, devops},
  note      = {ISBN: 978-0-321-60191-9},
  file      = {~/org/sources/books/Continuous Delivery_ Reliable Software Releases through -- Humble, Jez & Farley, David.pdf}
}
```

---

## 9. Magazine article (no abstract)

,**Situation**: Article in a popular-science or trade magazine (Scientific American,
IEEE Spectrum, Communications of the ACM magazine section, etc.). No formal abstract
printed - the article just begins. No DOI visible on pages read.

,**Rule**: Use `@article`. Set `journal` to the magazine name as printed. Include
`volume`, `number`, `pages`, `year`. Omit `abstract` entirely - do not summarize
or paraphrase. Omit `doi` if not found on pages read.

```bibtex
@article{schwartz2004tyranny-choice,
  author    = {Schwartz, Barry},
  title     = {The Tyranny of Choice},
  journal   = {Scientific American},
  volume    = {92},
  number    = {4},
  pages     = {70--75},
  year      = {2004},
  keywords  = {choice, decision-making, cognitive-load, paradox-of-choice, well-being},
  file      = {~/org/sources/papers/schwartz2004tyranny-of-choice.pdf}
}
```

---

## 10. Non-journal in `journal` field

,**Situation**: The `journal` field contains something that is clearly not a
journal - a company name, a personal website, an institution name, etc. (e.g.
`journal = {Lingenic LLC}`).

,**Rule**: Reconsider the `@type`. This is likely `@techreport` (if from a
named institution) or `@misc`. Move the institution to the `institution` field
for `@techreport`, or to `note` for `@misc`.

```bibtex
@misc{slavenskoj2026knuth-pseudocode,
  author      = {Slavenskoj, Danslav},
  title       = {Towards a Generalization of Knuth's Pseudocode Architecture:
                 From Algorithms to Knowledge},
  year        = {2026},
  note        = {Lingenic LLC},
  keywords    = {pseudocode, knowledge-representation, formal-systems, notation, llm},
  abstract    = {<the work's own abstract, copied verbatim and complete>},
  file        = {~/org/sources/knuth-pseudocode-generalization.pdf}
}
```

# Abstract Extraction Patterns

This file documents which patterns work for which PDF types, and records failures.
When `extract_abstract.py` fails on a new PDF layout, diagnose the failure,
add a new pattern to `extract_abstract.py`, and append a record here.

---

## How to self-heal a FAIL

1. Run `pdftotext -f 1 -l 2 <pdf> -` and inspect the raw text around the abstract
2. Identify why no pattern matched:
   - Different delimiter? (e.g. "Abstract." instead of "Abstract—")
   - Abstract starts on page 3?
   - No "Abstract" keyword at all (structured differently)?
3. Add a new pattern to the `patterns` list in `extract_abstract.py`
   - Give it a descriptive name
   - Keep it before the fallback patterns
   - Test it: `python3 extract_abstract.py <pdf>` should print `ABSTRACT:...`
4. Append a record below: pattern name, what it fixes, example PDF

#!/usr/bin/env python3
"""
Fetch structured metadata from CrossRef API by DOI.

Usage:
    python3 fetch_doi.py <doi>

Output (stdout, one per line):
    FIELD:key=value   - metadata fields from CrossRef
    NOTFOUND:<doi>    - DOI not in CrossRef
    ERROR:<reason>    - network or parse error

Fields returned (when available):
    type        - CrossRef work type (journal-article, proceedings-article, etc.)
    title
    author      - "Lastname, Firstname and Lastname, Firstname" format
    year
    journal     - container title (for articles)
    booktitle   - container title (for proceedings)
    volume
    number
    pages
    publisher
    doi
    abstract    - present for many publishers; JATS markup converted to plain text

Inference reads these fields and formats the final BibTeX entry
following examples.md conventions. fetch_doi.py only fetches - it does
not format.

Exit codes:
    0 - success (fields printed)
    1 - not found or error
"""

import sys
import html
import json
import re
import urllib.request
import urllib.error


def fetch_crossref(doi):
    url = f"https://api.crossref.org/works/{urllib.request.quote(doi, safe='/')}"
    req = urllib.request.Request(
        url,
        headers={"User-Agent": "citation-add/1.0"}
    )
    try:
        with urllib.request.urlopen(req, timeout=10) as resp:
            return json.loads(resp.read())
    except urllib.error.HTTPError as e:
        if e.code == 404:
            return None
        raise RuntimeError(f"HTTP {e.code}")
    except Exception as e:
        raise RuntimeError(str(e))


def strip_html(s):
    return re.sub(r"<[^>]+>", "", s or "").strip()


def jats_abstract(s):
    """CrossRef abstracts are JATS XML. Drop a leading "Abstract" heading, keep
    structured-abstract section titles as "Title:", keep paragraph breaks as
    spaces, and decode entities, so the text matches the printed abstract."""
    if not s:
        return ""
    s = re.sub(r"<jats:title>\s*(abstract|summary)\s*</jats:title>", " ", s, flags=re.I)
    s = re.sub(r"<jats:title>\s*([^<]*?)\s*:?\s*</jats:title>", r" \1: ", s)
    s = re.sub(r"</?jats:(p|sec)[^>]*>", " ", s)
    s = html.unescape(strip_html(s))
    s = re.sub(r"\s+", " ", s).strip()
    return re.sub(r"^(abstract|summary)\s*[:.\u2014-]?\s+", "", s, flags=re.I)


def parse_fields(data, doi):
    w = data.get("message", {})
    fields = {}

    fields["type"] = w.get("type", "")

    titles = w.get("title", [])
    if titles:
        fields["title"] = strip_html(titles[0])

    authors = w.get("author", [])
    if authors:
        parts = []
        for a in authors:
            if a.get("family"):
                given = a.get("given", "")
                parts.append(f"{a['family']}, {given}" if given else a["family"])
        if parts:
            fields["author"] = " and ".join(parts)

    for date_field in ("published-print", "published-online", "published", "issued"):
        dp = w.get(date_field, {}).get("date-parts", [[]])
        if dp and dp[0]:
            fields["year"] = str(dp[0][0])
            break

    containers = w.get("container-title", [])
    if containers:
        cr_type = fields.get("type", "")
        if cr_type == "proceedings-article":
            fields["booktitle"] = containers[0]
        else:
            fields["journal"] = containers[0]

    for f in ("volume", "issue", "page", "publisher"):
        v = w.get(f, "")
        if v:
            out_key = "number" if f == "issue" else ("pages" if f == "page" else f)
            fields[out_key] = str(v)

    fields["doi"] = doi

    abstract = jats_abstract(w.get("abstract", ""))
    if abstract:
        fields["abstract"] = abstract

    return fields


def main():
    if len(sys.argv) < 2:
        print("Usage: fetch_doi.py <doi>", file=sys.stderr)
        sys.exit(1)

    doi = sys.argv[1].strip()
    # Normalize - strip URL prefix if present
    doi = re.sub(r"^https?://doi\.org/", "", doi)
    doi = re.sub(r"^doi:\s*", "", doi, flags=re.IGNORECASE)

    try:
        data = fetch_crossref(doi)
    except RuntimeError as e:
        print(f"ERROR:{e}")
        sys.exit(1)

    if data is None:
        print(f"NOTFOUND:{doi}")
        sys.exit(1)

    fields = parse_fields(data, doi)
    for k, v in fields.items():
        # Escape newlines in multi-line values (abstract)
        v_clean = v.replace("\n", " ").replace("\r", "")
        print(f"FIELD:{k}={v_clean}")

    # Warn if CrossRef returned too few fields to build a complete entry
    # (excludes 'type' and 'doi' which are always present)
    substantive = {k for k in fields if k not in ('type', 'doi')}
    if len(substantive) < 4:
        missing = {'title', 'author', 'year', 'journal', 'booktitle'} - substantive
        print(f"SPARSE:CrossRef returned only {len(substantive)} substantive fields. "
              f"Missing: {', '.join(sorted(missing))}. Read PDF pages for fallback.")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Deterministically extract abstracts from PDFs using pdftotext + heuristics.

Usage:
    python3 extract_abstract.py <pdf_path>

Output (to stdout):
    ABSTRACT:<text>        - abstract found verbatim
    NOABSTRACT:<reason>    - no abstract expected (book, magazine, etc.)
    FAIL:<reason>          - extraction failed, fall back to inference

Exit codes:
    0 - ABSTRACT found
    1 - NOABSTRACT (not an error, just not applicable)
    2 - FAIL (extraction failed, fall back to inference)

Patterns file:
    patterns.md - documents what patterns work and what failed, for self-healing.
    When a new pattern is added to this script, append a note to patterns.md.
"""

import sys
import os
import re
import subprocess

PDF_PATH = os.path.expanduser(sys.argv[1]) if len(sys.argv) > 1 else None

# --- Helpers ---

def extract_text(pdf_path, pages="1-2"):
    """Run pdftotext on given page range, return text or None."""
    try:
        result = subprocess.run(
            ["pdftotext", "-f", pages.split("-")[0], "-l", pages.split("-")[-1], pdf_path, "-"],
            capture_output=True, text=True, timeout=30
        )
        return result.stdout if result.returncode == 0 else None
    except Exception:
        return None


def normalize(text):
    """Collapse whitespace."""
    return re.sub(r'\s+', ' ', text).strip()


# --- Pattern library ---
# Each pattern is tried in order. First match wins.
# Returns (abstract_text, pattern_name) or None.

def try_patterns(text):
    patterns = [
        # Pattern 1: IEEE/ACM two-column format
        # "Abstract—" at start; terminates at Index Terms, Keywords, Introduction heading,
        # or a blank line (two-column layout where body text follows after blank line).
        # Note: pdftotext sometimes splits "INTRODUCTION" as "I NTRODUCTION" due to column layout.
        ("ieee_abstract_dash",
         r'Abstract[—\-–]\s*(.+?)(?=\n\n|\n\s*(?:Index Terms|Keywords|I\.\s+I.?NTRODUCTION|\d+\.\s+INTRODUCTION))',
         re.DOTALL | re.IGNORECASE),

        # Pattern 2: ICLR/NeurIPS spaced-cap heading "A BSTRACT"
        # pdftotext splits the heading due to column layout: "A BSTRACT\n<text>"
        # Terminates at "I NTRODUCTION" (same split artifact) or blank line.
        ("iclr_a_bstract",
         r'A BSTRACT\s*\n(.+?)(?=\n\n|\n\s*I\s+NTRODUCTION)',
         re.DOTALL),

        # Pattern 2b: MDPI style - "Abstract" heading followed by article-info sidebar,
        # then blank line, then actual abstract text.
        # e.g. "Abstract\n\nAcademic Editors: ...\nReceived: ...\n\n<actual abstract>"
        ("mdpi_abstract_after_info",
         r'Abstract\s*\n\n(?:Academic Editors?:.*?\n|Received:.*?\n|Revised:.*?\n|Accepted:.*?\n)+\n(.+?)(?=\n\n|\nKeywords?:)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 3: "Abstract" as heading, body follows on next line(s)
        # Terminates at Keywords, Index Terms, numbered section, or end of text block.
        # Skip if immediately followed by article metadata (handled by mdpi_abstract_after_info).
        ("heading_abstract",
         r'(?<!\w)Abstract\s*\n\s*(.+?)(?=\n\s*(?:Keywords|Index Terms|1\s+Introduction|\d\.\s+[A-Z])|\n\n)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 4: "ABSTRACT" all caps inline - text follows on same line
        # e.g. "ABSTRACT As software systems..." (ACM/IEEE single-column)
        # Terminates at CCS Concepts, Keywords, or double newline.
        ("caps_abstract_inline",
         r'\bABSTRACT\s+(.+?)(?=\n\s*(?:CCS Concepts|Keywords|Additional Key Words|\d+\s+INTRODUCTION)|\n\n)',
         re.DOTALL),

        # Pattern 4b: "Abstract" title-case inline - text follows on same line
        # e.g. "Abstract Mining Software Repositories..." (Springer/arXiv preprints)
        # Terminates at Keywords, double newline, numbered section, or author email block.
        ("abstract_titlecase_inline",
         r'\bAbstract\s+(.+?)(?=\nKeywords|\n\n|\n\d+\s+Introduction|\n\S+\s+·\s+\S+|\nE-mail:)',
         re.DOTALL),

        # Pattern 5: "ABSTRACT" all caps as heading, text on next line
        ("caps_abstract",
         r'\bABSTRACT\b\s*\n\s*(.+?)(?=\n\s*(?:KEYWORDS|KEY WORDS|INTRODUCTION|Keywords|1\s+\w)|\n\n)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 6: "Abstract:" with colon
        ("abstract_colon",
         r'\bAbstract:\s*(.+?)(?=\n\s*(?:Keywords|Index Terms|1\.\s|\d\.\s+\w)|\n\n)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 7: "Abstract." with period (ECSA/Springer LNCS format)
        # e.g. "Abstract. To establish good architectural practice..."
        # Terminates at Keywords: or next blank line after section ends.
        ("abstract_dot",
         r'\bAbstract\.\s+(.+?)(?=\n\s*(?:Keywords|Index Terms|1\s+Introduction)|\n\n)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 8: "Abstract - " with space-dash (conference proceedings variant)
        # e.g. "Abstract - The maturation of agentic AI..."
        # Terminates at "Keywords - " or double newline.
        ("abstract_space_dash",
         r'\bAbstract\s+-\s+(.+?)(?=\nKeywords\s+-|\n\n)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 9: Elsevier "A R T I C L E I N F O / A B S T R A C T" sidebar format
        # pdftotext renders as "A B S T R A C T\n\nKeywords:\n<kw>\n...\n\n<abstract text>"
        # The keywords sidebar appears first, then a blank line, then the abstract.
        ("elsevier_spaced_abstract",
         r'A B S T R A C T\s*\n\n(?:Keywords:[^\n]*\n(?:[^\n]+\n)*)\n(.+?)(?=\n\n|\Z)',
         re.DOTALL),

        # Pattern 9b: Elsevier sidebar where Keywords appear BEFORE "a b s t r a c t"
        # pdftotext renders article-info + keywords block, then "a b s t r a c t\n<text>"
        # e.g. "Keywords:\n...\n\na b s t r a c t\n<abstract text>\n\n1. Introduction"
        ("elsevier_spaced_abstract_kw_before",
         r'a b s t r a c t\s*\n(.+?)(?=\n\n|\n\s*\d+\.\s+Introduction|\Z)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 10: ACM preprints with no explicit Abstract heading
        # Abstract text follows author affiliations and copyright boilerplate,
        # terminated by "CCS Concepts:" or "Additional Key Words".
        ("acm_no_heading_ccs",
         r'(?:permission[^\n]+\n.{0,500}?)\n\n(.+?)(?=\nCCS Concepts|\nAdditional Key Words)',
         re.DOTALL | re.IGNORECASE),

        # Pattern 10b: ACM/arXiv papers where abstract is inline before "CCS Concepts:"
        # with no explicit Abstract heading and no permission block anchor.
        # The abstract is the last paragraph-length block before "CCS Concepts:".
        # Strategy: find the last \n\n<text>\nCCS Concepts: where text >= 150 chars and
        # contains no \n\n (single paragraph).
        ("acm_ccs_inline",
         r'\n\n([^\n]{20}[^\n]*(?:\n[^\n]+){2,})\nCCS Concepts:',
         re.DOTALL),

        # Pattern 12: Nature Scientific Data / similar journals
        # Abstract is inline text directly after author line, terminated by "Background & Summary"
        # or next section heading. No "Abstract" keyword present.
        ("nature_background_summary",
         r'\n\n(.+?)\n\nBackground\s*&\s*Summary',
         re.DOTALL),

        # Pattern 11: arXiv style - abstract block bounded by blank lines after "Abstract"
        ("arxiv_blank_lines",
         r'\bAbstract\b\s*\n\n(.+?)\n\n',
         re.DOTALL | re.IGNORECASE),

    ]

    for name, pattern, flags in patterns:
        for m in re.finditer(pattern, text, flags):
            candidate = normalize(m.group(1))
            # Sanity checks: must be at least 150 chars, not more than 6000
            # (150 excludes titles and graphical abstract captions)
            if 150 <= len(candidate) <= 6000:
                return candidate, name

    return None


# --- Book/non-abstract detection ---

BOOK_SIGNALS = [
    r'all rights reserved',
    r'isbn[:\s]',
    r'library of congress',
    r'printed in',
    r'addison.?wesley',
    r"o'reilly",
    r'mit press',
    r'springer',
    r'draft copy',
    r'copyright.*\d{4}.*by',
    r'table of contents',
]

MAG_SIGNALS = [
    r'scientific american',
    r'ieee spectrum',
    r'communications of the acm',
    r'acm queue',
]


def is_book(text_lower):
    hits = sum(1 for sig in BOOK_SIGNALS if re.search(sig, text_lower))
    return hits >= 2


def is_magazine(text_lower):
    hits = sum(1 for sig in MAG_SIGNALS if re.search(sig, text_lower))
    return hits >= 1 and not re.search(r'\babstract\b', text_lower)


# --- Entry point ---

def main():
    if not PDF_PATH:
        print("FAIL:no pdf path given")
        sys.exit(2)

    if not os.path.exists(PDF_PATH):
        print(f"FAIL:file not found: {PDF_PATH}")
        sys.exit(2)

    # First pass: pages 1-2
    text = extract_text(PDF_PATH, "1-2")
    if not text or len(text.strip()) < 20:
        print("FAIL:pdftotext returned empty text")
        sys.exit(2)

    # Try patterns before book/magazine detection - patterns take priority
    result = try_patterns(text)
    if result:
        abstract, pattern_name = result
        print(f"ABSTRACT:{abstract}")
        print(f"PATTERN:{pattern_name}", file=sys.stderr)
        sys.exit(0)

    # Second pass: pages 1-15 (catches tech reports, PhD theses with late abstract pages)
    text_wide = extract_text(PDF_PATH, "1-15")
    if text_wide and len(text_wide.strip()) > len(text.strip()):
        result = try_patterns(text_wide)
        if result:
            abstract, pattern_name = result
            print(f"ABSTRACT:{abstract}")
            print(f"PATTERN:{pattern_name}(wide)", file=sys.stderr)
            sys.exit(0)

    # No patterns matched - check if this is a book or magazine (no abstract expected)
    text_lower = text.lower()
    if is_book(text_lower):
        print("NOABSTRACT:appears to be a book (no abstract expected)")
        sys.exit(1)

    if is_magazine(text_lower):
        print("NOABSTRACT:appears to be a magazine article (no formal abstract)")
        sys.exit(1)

    print("FAIL:abstract section not found by any pattern")
    sys.exit(2)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
Classify a PDF for citation-add: document class, DOI, pdfinfo metadata, and
which pages to read for the bib fields or the abstract.

Copied from the retired cite-nav skill; record classification mistakes and
fixes in classify.md next to this script.

Usage:
    python3 classify_pdf.py --pdf <path> --task <task> [--bib-type <type>] [--hint <hint>]

Tasks:
    classify          - classify document, return CLASS only
    extract:abstract  - return range/strategy for abstract extraction
    extract:bib       - return range/strategy and field list for bib entry extraction

Output (stdout):
    CLASS:<name>           - document class inferred
    RANGE:<start>-<end>    - page range to read (extract tasks only)
    STRATEGY:<name>        - strategy used (extract tasks only)
    FIELDS:<f1,f2,...>     - bib fields to read off the pages (extract:bib only);
                             never abstract, which citation-add copies verbatim
    NOTEXT:<reason>        - pdftotext returned empty (scanned/encrypted)

Exit codes:
    0 - success
    2 - NOTEXT or classification failed
"""

import sys
import os
import re
import subprocess
import argparse

def pdfinfo(pdf_path):
    """Return dict of pdfinfo fields."""
    try:
        result = subprocess.run(['pdfinfo', pdf_path], capture_output=True, text=True, timeout=10)
        info = {}
        for line in result.stdout.splitlines():
            if ':' in line:
                k, _, v = line.partition(':')
                info[k.strip()] = v.strip()
        return info
    except Exception:
        return {}

def page_count(pdf_path):
    info = pdfinfo(pdf_path)
    try:
        return int(info.get('Pages', 0))
    except ValueError:
        return 0

def meta_signals(info):
    """
    Extract classification signals and bib fields from pdfinfo metadata.
    Returns (signals_dict, fields_dict).
      signals_dict: keys that classify() can use (arxiv, thesis, etc.)
      fields_dict:  bib fields found (title, author, keywords)
    """
    title = info.get('Title', '').strip()
    author = info.get('Author', '').strip()
    subject = info.get('Subject', '').strip()
    keywords = info.get('Keywords', '').strip()
    creator = info.get('Creator', '').strip()

    title_lower = title.lower()
    creator_lower = creator.lower()

    signals = {}
    if 'arxiv genpdf' in creator_lower or 'arxiv' in creator_lower:
        signals['arxiv'] = True
    if re.search(r'\b(phd|ph\.d|doctoral|master.s thesis|dissertation)\b', title_lower):
        signals['thesis'] = True
    if subject:
        signals['subject'] = subject.lower()

    fields = {}
    if title:
        fields['title'] = title
    if author:
        fields['author'] = author
    if keywords:
        fields['keywords'] = keywords

    return signals, fields

def extract_doi(pdf_path, info=None, text=None):
    """
    Extract DOI from pdfinfo metadata or p.1 text.
    Only matches DOIs that appear near a doi: anchor to avoid picking up
    reference list DOIs.
    Returns DOI string or None.
    """
    bare_doi = re.compile(r'\b10\.\d{4,9}/[^\s,;>"\])\}]+')
    # Anchored: doi: or doi.org/ prefix - these are the paper's own DOI
    anchored_doi = re.compile(
        r'(?:doi\.org/|doi:\s*|DOI:\s*)(10\.\d{4,9}/[^\s,;>"\])\}]+)',
        re.IGNORECASE
    )

    # Check pdfinfo fields first (always reliable)
    if info:
        for field in ('doi', 'DOI', 'Subject', 'Keywords'):
            val = info.get(field, '')
            m = bare_doi.search(val)
            if m:
                return m.group(0).rstrip('.')

    # Scan p.1 text - anchored only to avoid reference DOIs
    if text is None:
        text = extract_text(pdf_path, "1-1") or ""
    m = anchored_doi.search(text)
    if m:
        doi = m.group(1).rstrip('.')
        # Filter placeholder DOIs (e.g. ACM template 10.1145/nnnnnnn)
        if not re.search(r'[nN]{3,}', doi):
            return doi

    return None


def extract_text(pdf_path, pages="1-2"):
    try:
        start, end = pages.split('-')
        result = subprocess.run(
            ['pdftotext', '-f', start, '-l', end, pdf_path, '-'],
            capture_output=True, text=True, timeout=30
        )
        return result.stdout if result.returncode == 0 else None
    except Exception:
        return None

def classify(pdf_path, bib_type=None, hint=None, info=None):
    """
    Classify document into one of:
      journal-article, conference-paper, arxiv-preprint,
      book, book-chapter, thesis, tech-report, unknown

    Priority: bib-type arg > pdfinfo metadata signals > filename > p.1 text > page count
    """
    # Bib type mapping - resolve unambiguous types immediately
    if bib_type:
        t = bib_type.lower()
        if t == 'book':
            return 'book'
        if t in ('incollection', 'inbook', 'bookinbook'):
            return 'book-chapter'
        if t in ('mastersthesis', 'phdthesis'):
            return 'thesis'
        if t == 'techreport':
            return 'tech-report'
        # inproceedings, article, misc: refine below (may be arxiv-preprint)

    if info is None:
        info = pdfinfo(pdf_path)
    meta, _ = meta_signals(info)

    # pdfinfo metadata signals - cheap, before reading any page text
    if meta.get('arxiv'):
        return 'arxiv-preprint'
    if meta.get('thesis'):
        return 'thesis'

    pages = page_count(pdf_path) if not info else int(info.get('Pages', 0) or 0)
    text = extract_text(pdf_path, "1-1") or ""
    text_lower = text.lower()
    filename = os.path.basename(pdf_path).lower()

    # arXiv signals - check early, before all other heuristics.
    # arXiv papers mention university affiliations which falsely trigger thesis,
    # and "Published as a conference paper at ICLR" which triggers conference.
    if (re.search(r'arxiv:', text_lower)
            or re.search(r'\d{4}\.\d{4,5}v\d', filename)
            or 'arxiv' in (hint or '').lower()):
        return 'arxiv-preprint'

    # SSRN signals
    if (re.search(r'ssrn-\d+', filename)
            or re.search(r'ssrn|social science research network', text_lower)):
        return 'ssrn-preprint'

    # inproceedings early return - safe now that arXiv has been checked
    if bib_type and bib_type.lower() == 'inproceedings':
        return 'conference-paper'

    # Extracted chapters from this edited ebook carry its cover and editor list.
    if 'in their own words' in text_lower and re.search(r'\beditors\b', text_lower):
        return 'book-chapter'

    # Book signals - page count is the primary signal; text signals alone are
    # too noisy (ACM DL downloads embed "all rights reserved" in metadata)
    if pages > 150:
        return 'book'

    # Thesis signals - require strong explicit markers, not just substrings
    # ('master' alone matches 'masterclass', 'masterful', university dept names)
    if re.search(r'\b(phd|ph\.d\.?|doctoral|master\'?s thesis|dissertation)\b', text_lower):
        return 'thesis'

    # Tech report signals
    if re.search(r'technical report|cmu/sei|sei/cmu|nist |rand corporation', text_lower):
        return 'tech-report'

    # Conference vs journal
    if re.search(r'proceedings|conference|workshop|symposium|icse|esec|fse|ecsa|icsa', text_lower):
        return 'conference-paper'

    if re.search(r'journal|volume|issue|elsevier|springer|wiley|ieee trans|nature\.com|mdpi\.com|tandfonline|sage pub', text_lower):
        return 'journal-article'

    # Fallback by page count
    if pages <= 12:
        return 'conference-paper'
    if pages <= 30:
        return 'journal-article'

    return 'unknown'


def strategy_for(doc_class, task, pages):
    """
    Return (strategy_name, page_range_string) for a doc class + task.

    Strategies:
      full-read    - read the whole document
      front-read   - read first N pages (abstract always here)
      toc-first    - read TOC pages, then navigate
      term-narrow  - read second half of document
    """
    # Normalize task aliases
    if task == 'abstract':
        task = 'extract:abstract'

    if task == 'extract:abstract':
        if doc_class in ('conference-paper', 'arxiv-preprint', 'ssrn-preprint'):
            return 'front-read', '1-2'
        if doc_class == 'journal-article':
            return 'front-read', '1-3'
        if doc_class == 'tech-report':
            return 'front-read', '1-15'   # SEI reports abstract on p9-11
        if doc_class in ('book', 'thesis'):
            return 'toc-first', f'1-{min(15, pages)}'
        return 'front-read', '1-3'

    if task == 'extract:bib':
        # Bib extraction needs the same range as abstract plus p.1 for metadata
        # For short docs, full-read is cheap enough
        if pages <= 30:
            return 'full-read', f'1-{pages}'
        if doc_class in ('book', 'thesis'):
            return 'toc-first', f'1-{min(15, pages)}'
        return 'front-read', '1-5'

    return 'full-read', f'1-{min(pages, 20)}'


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--pdf', required=True)
    parser.add_argument('--task', required=True)
    parser.add_argument('--bib-type', default=None)
    parser.add_argument('--hint', default=None)
    args = parser.parse_args()

    pdf_path = os.path.expanduser(args.pdf)
    if not os.path.exists(pdf_path):
        print(f"NOTEXT:file not found: {pdf_path}")
        sys.exit(2)

    # Quick text check
    text = extract_text(pdf_path, "1-1")
    if not text or len(text.strip()) < 20:
        print("NOTEXT:pdftotext returned empty (scanned or encrypted)")
        sys.exit(2)

    info = pdfinfo(pdf_path)
    pages = int(info.get('Pages', 0) or 0) or 1
    doc_class = classify(pdf_path, bib_type=args.bib_type, hint=args.hint, info=info)

    print(f"CLASS:{doc_class}")

    if args.task == 'classify':
        return

    strategy, page_range = strategy_for(doc_class, args.task, pages)
    print(f"RANGE:{page_range}")
    print(f"STRATEGY:{strategy}")

    if args.task == 'extract:bib':
        # Emit metadata fields found directly from pdfinfo
        _, meta_fields = meta_signals(info)
        for k, v in meta_fields.items():
            print(f"META:{k}={v}")
        # DOI - try pdfinfo then p.1 text
        doi = extract_doi(pdf_path, info=info)
        if doi:
            print(f"DOI:{doi}")
        # Fields still to read off the pages (those not in metadata or doi).
        # The abstract is not one of them: citation-add copies it verbatim from
        # CrossRef, arXiv or the PDF, or omits it.
        needed = [f for f in ('author', 'title', 'year', 'venue')
                  if f not in meta_fields]
        if not doi and doc_class not in ('book', 'thesis'):
            needed.append('doi')
        print(f"FIELDS:{','.join(needed)}")


if __name__ == '__main__':
    main()

#!/usr/bin/env python3
"""
Duplicate detection for citation-add.

Given a new PDF's extracted DOI and/or title, checks references.bib for
existing entries that may be the same work. Outputs structured instructions
for the calling inference agent.

Usage:
    python3 check_duplicate.py --bib ~/org/bib/references.bib
                               [--doi <doi>]
                               [--title <title>]
                               [--author <author>]
                               [--year <year>]
                               [--new-pdf <filename>]

Output (stdout, one per line):
    NO_MATCH
        No existing entry looks like this paper. Proceed to add.

    KEEP_EXISTING: citekey=<key> existing_file=<path> reason=<text>
        Deterministic: delete new PDF, keep existing. Reason explains why.

    KEEP_NEW: citekey=<key> existing_file=<path> reason=<text>
        Deterministic: update bib file field to new PDF, delete old PDF.

    INFER: citekey=<key> existing_file=<path> new_pdf=<filename> hint=<text>
        Read pages 1-2 of both PDFs and reason about whether they are the
        same work. Resolve as KEEP_EXISTING, KEEP_NEW, or ASK based on
        content. If resolved silently, log the decision in the run report.

    ASK: citekey=<key> existing_file=<path> message=<text>
        Genuinely ambiguous. Show message to user and wait for input.

Exit codes:
    0 - output produced (check first token to interpret)
    1 - usage error
"""

import sys
import re
import os
import argparse


ARXIV_RE = re.compile(r'(\d{4}\.\d{4,5})(?:v(\d+))?(?:\.pdf)?$', re.IGNORECASE)


def extract_field(name, block):
    """Extract a bib field value, handling nested braces correctly."""
    m = re.search(r'\b' + name + r'\s*=\s*\{', block, re.IGNORECASE)
    if not m:
        return ''
    start = m.end()
    depth = 1
    i = start
    while i < len(block) and depth > 0:
        if block[i] == '{':
            depth += 1
        elif block[i] == '}':
            depth -= 1
        i += 1
    return block[start:i - 1].strip()


def load_bib(bib_path):
    """Parse bib into list of dicts with keys: citekey, doi, title, author, year, file."""
    entries = []
    try:
        with open(bib_path, encoding='utf-8') as stream:
            text = stream.read()
    except FileNotFoundError:
        return entries

    for block in re.split(r'(?=@\w+\s*\{)', text):
        block = block.strip()
        if not block.startswith('@'):
            continue
        m = re.match(r'@\w+\s*\{([^,]+),', block)
        if not m:
            continue
        citekey = m.group(1).strip()

        entries.append({
            'citekey': citekey,
            'doi': extract_field('doi', block).lower().strip(),
            'title': extract_field('title', block).lower().strip(),
            'author': extract_field('author', block),
            'year': extract_field('year', block).strip(),
            'file': extract_field('file', block),
        })
    return entries


def normalize_doi(doi):
    if not doi:
        return ''
    doi = re.sub(r'^https?://doi\.org/', '', doi, flags=re.IGNORECASE)
    doi = re.sub(r'^doi:\s*', '', doi, flags=re.IGNORECASE)
    return doi.lower().strip()


def title_words(title):
    """Return set of significant lowercase words from a title."""
    stop = {'a', 'an', 'the', 'of', 'in', 'on', 'for', 'and', 'or', 'to',
            'with', 'using', 'towards', 'combining', 'via'}
    words = re.findall(r'[a-z0-9]+', (title or '').lower())
    return {w for w in words if w not in stop and len(w) > 2}


def author_surnames(author_str):
    """Return set of lowercase surnames from a bib author string.

    Handles both 'Lastname, Firstname and ...' and 'Firstname Lastname and ...' forms.
    Also handles raw nav.py output like 'John Smith; Jane Doe'.
    """
    if not author_str:
        return set()
    # Split on ' and ' or ';'
    parts = re.split(r'\s+and\s+|;', author_str, flags=re.IGNORECASE)
    surnames = set()
    for part in parts:
        part = part.strip()
        if not part:
            continue
        # 'Lastname, Firstname' form
        if ',' in part:
            surname = part.split(',')[0].strip()
        else:
            # 'Firstname Lastname' - take last token
            tokens = part.split()
            surname = tokens[-1] if tokens else ''
        # Strip LaTeX accents like {\v{c}} -> just take alpha chars
        surname = re.sub(r'\{[^}]*\}', '', surname)
        surname = re.sub(r'[^a-zA-Z]', '', surname).lower()
        if surname:
            surnames.add(surname)
    return surnames


def arxiv_id_and_version(filename):
    """Return (arxiv_id, version_int) or (None, None)."""
    m = ARXIV_RE.search(filename or '')
    if m:
        arxiv_id = m.group(1)
        version = int(m.group(2)) if m.group(2) else 0
        return arxiv_id, version
    return None, None


def signals_match(new_title, new_author, new_year, entry, title_threshold):
    """Check whether title, author, and year all agree with an existing entry.

    Returns (matched: bool, overlap: float, missing_signals: list).
    missing_signals lists which signals were unavailable (not failed).
    All available signals must match; missing signals are noted in the hint.
    """
    missing = []

    # Title
    new_words = title_words(new_title)
    existing_words = title_words(entry['title'])
    if not new_words or not existing_words:
        return False, 0.0, missing
    overlap = len(new_words & existing_words) / max(len(new_words), len(existing_words))
    if overlap < title_threshold:
        return False, overlap, missing

    # Author
    new_surnames = author_surnames(new_author)
    existing_surnames = author_surnames(entry['author'])
    if new_surnames and existing_surnames:
        if not (new_surnames & existing_surnames):
            return False, overlap, missing
    else:
        missing.append('author')

    # Year
    new_yr = (new_year or '').strip()
    existing_yr = (entry['year'] or '').strip()
    if new_yr and existing_yr:
        if new_yr != existing_yr:
            return False, overlap, missing
    else:
        missing.append('year')

    return True, overlap, missing


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--bib', required=True)
    parser.add_argument('--doi', default='')
    parser.add_argument('--title', default='')
    parser.add_argument('--author', default='')
    parser.add_argument('--year', default='')
    parser.add_argument('--new-file', '--new-pdf', dest='new_pdf', default='')
    args = parser.parse_args()

    new_doi = normalize_doi(args.doi)
    new_title = args.title.lower().strip()
    new_author = args.author
    new_year = args.year.strip()
    new_pdf = args.new_pdf
    entries = load_bib(args.bib)

    new_arxiv_id, new_arxiv_ver = arxiv_id_and_version(new_pdf)

    for e in entries:
        citekey = e['citekey']
        existing_file = e['file']

        # --- Case 1: exact DOI match (deterministic) ---
        if new_doi and e['doi'] and new_doi == e['doi']:
            new_format = os.path.splitext(new_pdf)[1].lower()
            existing_formats = {'.' + suffix.lower() for suffix in re.findall(r'\.(pdf|epub)(?=[;:\s]|$)', existing_file, re.I)}
            if new_format in {'.pdf', '.epub'} and existing_formats and new_format not in existing_formats:
                print(f"ADD_FORMAT: citekey={citekey} existing_file={existing_file} new_file={new_pdf} reason=Same DOI, additional source format")
                return
            print(f"KEEP_EXISTING: citekey={citekey} existing_file={existing_file} "
                  f"reason=DOI match ({new_doi})")
            return

        # --- Case 2: arXiv ID match (deterministic version comparison) ---
        existing_arxiv_id, existing_arxiv_ver = arxiv_id_and_version(existing_file)
        if new_arxiv_id and existing_arxiv_id and new_arxiv_id == existing_arxiv_id:
            if new_arxiv_ver > existing_arxiv_ver:
                print(f"KEEP_NEW: citekey={citekey} existing_file={existing_file} "
                      f"reason=arXiv {new_arxiv_id} v{new_arxiv_ver} > v{existing_arxiv_ver}")
            else:
                print(f"KEEP_EXISTING: citekey={citekey} existing_file={existing_file} "
                      f"reason=arXiv {new_arxiv_id} existing v{existing_arxiv_ver} >= new v{new_arxiv_ver}")
            return

        # --- Case 3: new has DOI, existing is arXiv preprint (published beats preprint) ---
        if new_doi and existing_arxiv_id and not e['doi']:
            matched, overlap, missing = signals_match(
                new_title, new_author, new_year, e, title_threshold=0.6
            )
            if matched:
                caveat = f" (unconfirmed: {', '.join(missing)})" if missing else ""
                print(f"INFER: citekey={citekey} existing_file={existing_file} "
                      f"new_pdf={new_pdf} "
                      f"hint=New PDF has DOI {new_doi}, existing is arXiv preprint. "
                      f"Title overlap {overlap:.0%}, author+year match{caveat}. "
                      f"Read p1-2 of each: if same work, keep new (published) and delete arXiv file.")
                return

        # Different DOIs - definitely not a duplicate
        if new_doi and e['doi'] and new_doi != e['doi']:
            continue

        # --- Case 4: title + author + year similarity (all three required) ---
        if new_title and e['title']:
            matched, overlap, missing = signals_match(
                new_title, new_author, new_year, e, title_threshold=0.7
            )
            if matched:
                caveat = f" (unconfirmed: {', '.join(missing)})" if missing else ""
                print(f"INFER: citekey={citekey} existing_file={existing_file} "
                      f"new_pdf={new_pdf} "
                      f"hint=Title overlap {overlap:.0%}, author+year match{caveat}. "
                      f"No DOI confirmation - read p1-2 of each to verify same work.")
                return

    print("NO_MATCH")


if __name__ == '__main__':
    main()

"""Shared catalog and path rules for the local source library."""
from pathlib import Path
import os
import re

class Field:
    def __init__(self, name, value, vstart, vend, line_start, line_end, delim):
        self.name = name  # lowercased
        self.value = value  # raw text between delimiters
        self.vstart, self.vend = vstart, vend  # span of value (inside delimiters)
        self.line_start, self.line_end = line_start, line_end  # whole field incl. comma+newline
        self.delim = delim


class Entry:
    def __init__(self, etype, key, start, end):
        self.etype, self.key, self.start, self.end = etype, key, start, end
        self.fields = {}

    def get(self, name):
        f = self.fields.get(name)
        return f.value if f else None


def _match_brace(text, i):
    """text[i] == '{'; return index of the matching '}'."""
    depth = 0
    n = len(text)
    while i < n:
        c = text[i]
        if c == "\\":
            i += 2
            continue
        if c == "{":
            depth += 1
        elif c == "}":
            depth -= 1
            if depth == 0:
                return i
        i += 1
    raise ValueError("unbalanced braces")


def _delete_span(text, name_pos, after):
    """Span to cut so the field disappears and the layout around it stays intact.

    A field alone on its line(s) loses the whole line; a field sharing a line
    with others loses only itself, its comma and the spaces next to it.
    """
    ls = text.rfind("\n", 0, name_pos) + 1
    alone_before = text[ls:name_pos].strip() == ""
    cm = re.compile(r"[ \t]*,[ \t]*").match(text, after)
    end = cm.end() if cm else after
    at_eol = text[end : end + 1] == "\n"
    if alone_before and at_eol:
        return ls, end + 1
    if alone_before:
        return name_pos, end
    start = name_pos
    if at_eol:
        while start > 0 and text[start - 1] in " \t":
            start -= 1
    return start, end


def parse_bib(text):
    entries = []
    for m in re.finditer(r"(?m)^@(\w+)\s*\{", text):
        etype = m.group(1).lower()
        if etype in ("comment", "string", "preamble"):
            continue
        open_i = m.end() - 1
        close_i = _match_brace(text, open_i)
        km = re.compile(r"\s*([^,\s]+)\s*,").match(text, m.end())
        e = Entry(etype, km.group(1), m.start(), close_i + 1)
        i = km.end()
        while True:
            fm = re.compile(r"\s*([A-Za-z][\w-]*)\s*=\s*").match(text, i)
            if not fm or fm.end() > close_i:
                break
            name = fm.group(1).lower()
            j = fm.end()
            if text[j] == "{":
                vend = _match_brace(text, j)
                vstart, after, delim = j + 1, vend + 1, "{"
            elif text[j] == '"':
                k = j + 1
                depth = 0
                while True:
                    c = text[k]
                    if c == "\\":
                        k += 2
                        continue
                    if c == "{":
                        depth += 1
                    elif c == "}":
                        depth -= 1
                    elif c == '"' and depth == 0:
                        break
                    k += 1
                vstart, vend, after, delim = j + 1, k, k + 1, '"'
            else:
                bm = re.compile(r"[^,}\s]+").match(text, j)
                vstart, vend, after, delim = j, bm.end(), bm.end(), ""
            line_start, line_end = _delete_span(text, fm.start(1), after)
            e.fields[name] = Field(name, text[vstart:vend], vstart, vend, line_start, line_end, delim)
            i = after
            sm = re.compile(r"\s*,").match(text, i)
            if sm:
                i = sm.end()
        entries.append(e)
    return entries

SOURCE_ROOT = Path.home() / "org/sources"

              "incollection", "bookinbook", "suppbook", "suppcollection",
              "reference", "mvreference", "inreference", "booklet"}
PAPER_TYPES = {"article", "inproceedings", "conference", "unpublished"}

                 "magazine": "periodicals", "bookchapter": "books"}


def category_for(entry_type, entry_subtype=None):
    kind = entry_type.lower()
    if kind in BOOK_TYPES:
        return "books"
    if kind in PAPER_TYPES:
        return "papers"
    if kind in THESIS_TYPES:
        return "theses"
    if kind in REPORT_TYPES:
        return "reports"
    if kind in WEB_TYPES:
        return "web"
    if kind in PERIODICAL_TYPES:
        return "periodicals"
    if kind == "misc":
        return MISC_SUBTYPES.get((entry_subtype or "").lower())
    return None


def read_entries(bib=BIB_PATH):
    return parse_bib(Path(bib).expanduser().read_text(encoding="utf-8"))


def attachment_paths(entry, bib=BIB_PATH, root=SOURCE_ROOT):
    """Resolve simple and Zotero/Mendeley file fields without basename matching."""
    result = []
    for item in re.split(r"(?<!\\);", entry.get("file") or ""):
        item = item.strip()
        if not item:
            continue
        triplet = re.fullmatch(r"[^:]*:((?:/|~|\.{1,2}/).+):[A-Za-z0-9]+", item)
        if triplet:
            item = triplet.group(1)
        for value in re.split(r"(?<!\\):(?=(?:/|~|\.{1,2}/))", item):
            value = re.sub(r"\\([:;])", r"\1", value)
            path = Path(os.path.expandvars(value)).expanduser()
            if not path.is_absolute():
                candidates = [Path(root) / path, Path(bib).expanduser().parent / path]
                path = next((candidate for candidate in candidates if candidate.exists()), candidates[0])
            result.append(path.resolve())
    return result


def source_files(root=SOURCE_ROOT):
    return sorted(path for path in Path(root).expanduser().rglob("*")
                  if path.is_file() and path.suffix.lower() in SOURCE_EXTENSIONS)


def catalog_files(entries, bib=BIB_PATH, root=SOURCE_ROOT):
    indexed = {}
    for entry in entries:
        for path in attachment_paths(entry, bib, root):
            indexed.setdefault(path, []).append(entry)
    return indexed

#!/usr/bin/env python3
"""Read-only audit of source attachments, folder placement, and inbox items."""
import argparse
import json
from pathlib import Path

from source_library import (BIB_PATH, SOURCE_ROOT, attachment_paths,
                            catalog_files, category_for, read_entries, source_files)


def audit(root, bib):
    root, bib = Path(root).expanduser().resolve(), Path(bib).expanduser().resolve()
    entries = read_entries(bib)
    indexed = catalog_files(entries, bib, root)
    issues = []
    files = source_files(root)
    for path in files:
        matches = indexed.get(path.resolve(), [])
        relative = str(path.relative_to(root))
        if not matches:
            issues.append({"kind": "uncataloged", "file": relative})
            continue
        if len(matches) > 1:
            issues.append({"kind": "shared-attachment", "file": relative,
                           "citekeys": [entry.key for entry in matches]})
        categories = {category_for(entry.etype, entry.get("entrysubtype")) for entry in matches}
        if len(categories) != 1:
            issues.append({"kind": "conflicting-types", "file": relative})
            continue
        expected = categories.pop()
        if expected is None:
            issues.append({"kind": "needs-category", "file": relative,
                           "types": sorted({entry.etype for entry in matches}),
                           "citekeys": [entry.key for entry in matches]})
        # Original source scans remain alongside the organized reading copy.
        # They must still be cataloged, and missing originals fail below.
        elif path.relative_to(root).parts[0] == "originals":
            continue
        elif path.relative_to(root).parts[0] != expected:
            issues.append({"kind": "unfiled" if path.parent == root else "misplaced",
                           "file": relative, "expected": expected,
                           "citekeys": [entry.key for entry in matches]})
    for entry in entries:
        for path in attachment_paths(entry, bib, root):
            if not path.is_file():
                issues.append({"kind": "missing-attachment", "file": str(path),
                               "citekeys": [entry.key]})
    return {"root": str(root), "files": len(files), "entries": len(entries), "issues": issues}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--root", type=Path, default=SOURCE_ROOT)
    parser.add_argument("--bib", type=Path, default=BIB_PATH)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()
    report = audit(args.root, args.bib)
    if args.json:
        print(json.dumps(report, ensure_ascii=False, indent=2))
    else:
        print(f"{report['files']} PDF/EPUB files; {report['entries']} bibliography entries")
        for issue in report["issues"]:
            context = ", ".join(issue.get("citekeys", []))
            expected = f" -> {issue['expected']}/" if "expected" in issue else ""
            print(f"{issue['kind']}: {issue['file']}{expected}" + (f" [{context}]" if context else ""))
    # The root is an inbox. Pending classifications and new files are reports,
    # while broken paths and incorrect established placement fail the audit.
    return int(any(issue["kind"] in {"missing-attachment", "misplaced", "conflicting-types"}
                   for issue in report["issues"]))

    raise SystemExit(main())

"""Read EPUB package metadata and, optionally, its spine text using the standard library."""
import argparse
from html.parser import HTMLParser
import json
from pathlib import Path
import posixpath
from urllib.parse import unquote
import xml.etree.ElementTree as ET
from zipfile import ZipFile


class Text(HTMLParser):
    def __init__(self):
        super().__init__(convert_charrefs=True)
        self.parts = []
        self.hidden = 0

    def handle_starttag(self, tag, attrs):
        if tag in {"script", "style"}:
            self.hidden += 1

    def handle_endtag(self, tag):
        if tag in {"script", "style"}:
            self.hidden = max(0, self.hidden - 1)
        if tag in {"p", "div", "li", "h1", "h2", "h3", "section", "br"}:
            self.parts.append("\n")

    def handle_data(self, data):
        if not self.hidden:
            self.parts.append(data)


def inspect(path, include_text=False):
    with ZipFile(Path(path).expanduser()) as archive:
        container = ET.fromstring(archive.read("META-INF/container.xml"))
        rootfile = container.find(".//{*}rootfile")
        if rootfile is None or not rootfile.get("full-path"):
            raise ValueError("EPUB has no package document")
        package_path = rootfile.get("full-path")
        package = ET.fromstring(archive.read(package_path))
        metadata = package.find("{*}metadata")
        result = {"format": "epub", "metadata": {}}
        if metadata is not None:
            for item in metadata:
                if item.tag.startswith("{http://purl.org/dc/elements/1.1/}"):
                    result["metadata"].setdefault(item.tag.rsplit("}", 1)[1], []).append("".join(item.itertext()).strip())
        if include_text:
            manifest = {item.get("id"): item for item in package.findall("{*}manifest/{*}item")}
            result["sections"] = []
            for reference in package.findall("{*}spine/{*}itemref"):
                item = manifest.get(reference.get("idref"))
                if item is None or item.get("media-type") not in {"application/xhtml+xml", "text/html"}:
                    continue
                filename = posixpath.normpath(posixpath.join(posixpath.dirname(package_path), unquote(item.get("href").split("#", 1)[0])))
                reader = Text()
                reader.feed(archive.read(filename).decode("utf-8-sig"))
                result["sections"].append({"file": filename, "text": "".join(reader.parts).strip()})
        return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("file", type=Path)
    parser.add_argument("--text", action="store_true", help="Include XHTML content in spine order")
    args = parser.parse_args()
    print(json.dumps(inspect(args.file, args.text), ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()

Every abstract must be the work's own abstract, verbatim and complete, or be

--keys limits fetch, score and apply to those citekeys (new entries); without

Sources, cached under ~/.cache/citation-add/abstract-audit/:
    crossref  https://api.crossref.org/works/<doi>          (JATS converted)

LIGATURES = {"\ufb00": "ff", "\ufb01": "fi", "\ufb02": "fl", "\ufb03": "ffi", "\ufb04": "ffl", "\ufb05": "st", "\ufb06": "st"}
LATEX_ACCENT = re.compile(r"\\[`'^\"~=.uvHcdbtrk]\s*(?:\{\s*([A-Za-z])\s*\}|([A-Za-z]))")
LATEX_LETTERS = {r"\o": "o", r"\O": "O", r"\ss": "ss", r"\ae": "ae", r"\AE": "AE", r"\l": "l", r"\L": "L", r"\i": "i"}


def strip_latex(s):
    s = LATEX_ACCENT.sub(lambda m: m.group(1) or m.group(2), s)
    for k, v in LATEX_LETTERS.items():
        s = re.sub(re.escape(k) + r"(?![A-Za-z])", v, s)
    s = re.sub(r"\\(?:textit|textbf|emph|textsc|texttt|textrm|mathrm|url)\s*\{", "{", s)
    s = re.sub(r"\\([&%#$_{}])", r"\1", s)
    s = s.replace("{", "").replace("}", "")
    s = s.replace("~", " ").replace("--", "-")
    return s


def normalize(s):
    """Lowercased word tokens, robust to quoting, dashes, hyphenation, accents, LaTeX."""
    s = strip_latex(s)
    for k, v in LIGATURES.items():
        s = s.replace(k, v)
    s = s.replace("\u00ad", "")  # soft hyphen
    s = re.sub(r"[\u2018\u2019\u201a\u201b\u2032`]", "'", s)
    s = re.sub(r"[\u201c\u201d\u201e\u201f\u2033]", '"', s)
    s = re.sub(r"[\u2010\u2011\u2012\u2013\u2014\u2015\u2212]", "-", s)
    # rejoin words broken across lines, and treat in-word hyphens as joins,
    # so "exam-\nple", "data-\ndriven" and "data-driven" all compare equal
    s = re.sub(r"(\w)-\s*\n\s*(\w)", r"\1\2", s)
    s = re.sub(r"(\w)-(\w)", r"\1\2", s)
    s = re.sub(r"(\w)'(\w)", r"\1\2", s)
    s = unicodedata.normalize("NFKD", s)
    s = "".join(c for c in s if not unicodedata.combining(c))
    s = s.lower()
    s = re.sub(r"\s+", " ", s)
    return re.findall(r"[a-z0-9]+", s)


def grams(tokens, n=5):
    if not tokens:
        return set()
    if len(tokens) < n:
        return {tuple(tokens)}
    return {tuple(tokens[i : i + n]) for i in range(len(tokens) - n + 1)}


def containment(a_grams, b_grams):
    """Fraction of a's n-grams found in b."""
    if not a_grams:
        return 0.0
    return len(a_grams & b_grams) / len(a_grams)


def align_ratio(a, b, n=20):
    """Similarity of a to the stretch of b it most plausibly came from.

    Both sides are reduced to normalized letters with spacing removed, so PDF
    artifacts ("whic hthe", a dropped "fi" ligature) cost a few letters, while
    words or sentences dropped from the middle make the window longer than a
    and pull the ratio down. Returns 0.0 when no window is found.
    """
    A, B = "".join(normalize(a)), "".join(normalize(b))
    if len(A) < n or len(B) < n:
        return 0.0
    first = last = None
    for i in range(len(A) - n + 1):
        k = B.find(A[i : i + n])
        if k >= 0:
            first = max(0, k - i)
            break
    if first is None:
        return 0.0
    for j in range(len(A) - n, -1, -1):
        k = B.find(A[j : j + n], first)
        if k >= 0:
            last = k + n + (len(A) - j - n)
            break
    if last is None or last <= first or last - first > 3 * len(A):
        return 0.0
    W = B[first:last]
    sm = difflib.SequenceMatcher(None, A, W, autojunk=False)
    matched = sum(b.size for b in sm.get_matching_blocks() if b.size >= 4)
    return matched / max(len(A), len(W))


def title_sim(a, b):
    ta, tb = set(normalize(a or "")), set(normalize(b or ""))
    if not ta or not tb:
        return 0.0
    return len(ta & tb) / min(len(ta), len(tb))

class RateLimiter:
    def __init__(self, interval):
        self.interval, self.lock, self.last = interval, threading.Lock(), 0.0

    def wait(self):
        with self.lock:
            delay = self.last + self.interval - time.monotonic()
            if delay > 0:
                time.sleep(delay)
            self.last = time.monotonic()


LIMITS = {"crossref": RateLimiter(0.5), "openalex": RateLimiter(0.25), "arxiv": RateLimiter(3.5)}


def http_get(url, limiter, retries=3):
    for attempt in range(retries):
        limiter.wait()
        req = urllib.request.Request(url, headers={"User-Agent": UA})
        try:
            with urllib.request.urlopen(req, timeout=30) as r:
                return r.status, r.read().decode("utf-8", "replace")
        except urllib.error.HTTPError as e:
            if e.code in (404, 410):
                return e.code, ""
            if e.code in (429, 500, 502, 503, 504) and attempt < retries - 1:
                time.sleep(5 * (attempt + 1))
                continue
            return e.code, ""
        except Exception as e:  # network errors
            if attempt < retries - 1:
                time.sleep(5 * (attempt + 1))
                continue
            return -1, str(e)
    return -1, ""


def cache_path(kind, key):
    safe = re.sub(r"[^A-Za-z0-9._-]+", "_", key)
    d = os.path.join(CACHE, kind)
    os.makedirs(d, exist_ok=True)
    return os.path.join(d, safe + ".json")


def cached(kind, key, fn):
    p = cache_path(kind, key)
    if os.path.exists(p):
        with open(p, encoding="utf-8") as f:
            return json.load(f)
    val = fn()
    if val.get("status", 0) in (-1, 429) or (isinstance(val.get("status"), int) and val["status"] >= 500):
        return val  # do not cache transient failures
    with open(p, "w", encoding="utf-8") as f:
        json.dump(val, f, ensure_ascii=False, indent=1)
    return val


def jats_to_text(s):
    if not s:
        return ""
    # drop a leading "Abstract"/"Summary" heading; keep other section titles as words
    s = re.sub(r"<jats:title>\s*(abstract|summary|a b s t r a c t)\s*</jats:title>", " ", s, flags=re.I)
    s = re.sub(r"<jats:title>\s*([^<]*?)\s*</jats:title>", lambda m: " " + m.group(1).rstrip(":") + ": ", s)
    s = re.sub(r"</?jats:(p|sec)[^>]*>", " ", s)
    s = re.sub(r"<[^>]+>", "", s)
    s = html.unescape(s)
    s = re.sub(r"\s+", " ", s).strip()
    s = re.sub(r"^(abstract|summary)\s*[:.\-\u2014]?\s+", "", s, flags=re.I)
    return s

def fetch_arxiv(ids):
    """Fetch many arXiv ids; returns {id: {...}} and caches each id."""
    todo = [i for i in ids if not os.path.exists(cache_path("arxiv", i))]
    for k in range(0, len(todo), 50):
        batch = todo[k : k + 50]
        url = "https://export.arxiv.org/api/query?max_results=100&id_list=" + ",".join(batch)
        status, body = http_get(url, LIMITS["arxiv"])
        if status != 200:
            print(f"arxiv batch failed: {status}", file=sys.stderr)
            continue
        ns = {"a": "http://www.w3.org/2005/Atom"}
        root = ET.fromstring(body)
        found = {}
        for ent in root.findall("a:entry", ns):
            eid = ent.findtext("a:id", "", ns)
            m = re.search(r"abs/([^v\s]+?)(v\d+)?$", eid)
            if not m:
                continue
            found[m.group(1)] = {
                "status": 200,
                "title": re.sub(r"\s+", " ", ent.findtext("a:title", "", ns)).strip(),
                "abstract": re.sub(r"\s+", " ", ent.findtext("a:summary", "", ns)).strip(),
            }
        for i in batch:
            val = found.get(i, {"status": 404})
            with open(cache_path("arxiv", i), "w", encoding="utf-8") as f:
                json.dump(val, f, ensure_ascii=False, indent=1)
    out = {}
    for i in ids:
        with open(cache_path("arxiv", i), encoding="utf-8") as f:
            out[i] = json.load(f)
    return out

def pdf_cuts(text):
    """Candidate regex cuts of the abstract, best first: the text after an
    "Abstract" heading, ending at the next heading (Keywords, Index Terms,
    1 Introduction) when that is the same block, else at the first blank line or
    page break, else at a blank line across a page break, else at the heading.
    The add-bibs skill's pattern library comes last. Returns [(text, how)]."""
    out = []
    words = lambda t: len(re.findall(r"\w+", t or ""))
    for m in CUT_START.finditer(text):
        rest = text[m.end() :].lstrip()
        he = CUT_END.search(rest)
        head_cut = rest[: he.start()] if he else None
        pb = re.search(r"\n[ \t\f]*\n|\f", rest)
        para_cut = rest[: pb.start()] if pb else None
        pb2 = re.search(r"\n[ \t]*\n", rest)
        para2_cut = rest[: pb2.start()] if pb2 else None
        if head_cut and para_cut and len(head_cut) <= 1.15 * len(para_cut) and words(head_cut) >= 40:
            out.append((head_cut, "heading"))
        if para_cut and words(para_cut) >= 40:
            after = rest[pb.end() :].lstrip()
            nb = re.search(r"\n[ \t\f]*\n|\f", after)
            nxt = after[: nb.start()] if nb else after
            prose = words(nxt) >= 30 and not CUT_END.match("\n" + nxt) and re.search(r"[.?!]\s*$", nxt.strip())
            out.append((para_cut, "paragraph+prose" if prose else "paragraph"))
        if para2_cut and para2_cut != para_cut and words(para2_cut) >= 40:
            out.append((para2_cut, "paragraph-across-page"))
        if head_cut and 40 <= words(head_cut) <= 700:
            out.append((head_cut, "heading-long"))
    if _skill_patterns:
        r = _skill_patterns(text)
        if r:
            out.append((r[0], "skill:" + r[1]))
    seen, uniq = set(), []
    for t, how in out:
        t = t.strip()
        if t not in seen:
            seen.add(t)
            uniq.append((t, how))
    return uniq


def ocr_pdf(path, first=None, last=None):
    """OCR a scanned PDF with pdftoppm + tesseract; pages joined with form feeds."""
    import tempfile

    out = []
    with tempfile.TemporaryDirectory() as d:
        cmd = ["pdftoppm", "-r", "300", "-gray", "-png"]
        if first:
            cmd += ["-f", str(first)]
        if last:
            cmd += ["-l", str(last)]
        subprocess.run(cmd + [path, os.path.join(d, "p")], capture_output=True, timeout=900)
        for img in sorted(os.listdir(d)):
            r = subprocess.run(["tesseract", os.path.join(d, img), "-", "--psm", "1"], capture_output=True, timeout=300)
            out.append(r.stdout.decode("utf-8", "replace"))
    return "\f".join(out)


def fetch_pdf(key, path):
    # An EPUB attachment must never be sent through the PDF/OCR pipeline.
    from source_library import attachment_paths
    pdfs = [str(p) for p in attachment_paths({"file": path}) if p.suffix.lower() == ".pdf"]
    if not pdfs:
        return {"status": 415, "path": path, "reason": "No PDF attachment"}

        return {"status": 200, "path": path, "pages": pdf_info(p), "ocr": ocr, "full": full, "head": head, "wide": wide}

    return cached("pdf", key, go)


def search_doi(e):
    """Find a DOI for an entry that has none: CrossRef bibliographic query, accepted
    only on an exact normalized title match (or main title before a colon) and a
    year within one. The bib's own fields are never changed by this."""
    title = strip_latex(e.get("title") or "")
    author = strip_latex((e.get("author") or e.get("editor") or "").split(" and ")[0].split(",")[0])
    year = re.sub(r"\D", "", e.get("year") or "")

    def go():
        q = {"query.bibliographic": title, "rows": "5", "select": "DOI,title,issued,type"}
        if author:
            q["query.author"] = author
        url = "https://api.crossref.org/works?" + urllib.parse.urlencode(q)
        status, body = http_get(url, LIMITS["crossref"])
        out = {"status": status, "items": []}
        if status == 200:
            for it in json.loads(body).get("message", {}).get("items", []):
                y = ((it.get("issued") or {}).get("date-parts") or [[None]])[0][0]
                out["items"].append({"doi": it.get("DOI"), "title": (it.get("title") or [""])[0], "year": y, "type": it.get("type")})
        return out

    r = cached("search", e.key, go)
    want = normalize(title)
    want_main = normalize(title.split(":")[0])
    for it in r.get("items", []):
        got = normalize(it["title"])
        if got != want and got != want_main and normalize(it["title"].split(":")[0]) != want:
            continue
        if year and it.get("year") and abs(int(it["year"]) - int(year[:4])) > 1:
            continue
        return it["doi"]
    return None

ARXIV_RE = re.compile(r"(?:arxiv[.:/ ]+(?:abs/|pdf/)?|10\.48550/arxiv\.)(\d{4}\.\d{4,5}|[a-z-]+(?:\.[A-Z]{2})?/\d{7})", re.I)


def arxiv_id(e):
    doi = (e.get("doi") or "").strip()
    m = re.match(r"10\.48550/arxiv\.(.+)$", doi, re.I)
    if m:
        return re.sub(r"v\d+$", "", m.group(1))
    if (e.get("eprinttype") or "").lower() == "arxiv" and e.get("eprint"):
        return re.sub(r"v\d+$", "", e.get("eprint").strip())
    for f in ("url", "journal", "note", "howpublished"):
        v = e.get(f) or ""
        m = ARXIV_RE.search(v)
        if m:
            return re.sub(r"v\d+$", "", m.group(1))
    return None

def entry_doi(e, search=False):
    """(doi, how): the bib's own DOI, else one found by title search (cached),

    return None, None

_DICT = None


def dict_word(w):
    """True if w (or w minus a common suffix) is in the system word list."""
    global _DICT
    if _DICT is None:
        try:
            with open("/usr/share/dict/words") as f:
                _DICT = {x.strip().lower() for x in f}
        except OSError:
            _DICT = set()
    w = w.lower()
    if w in _DICT:
        return True
    for suf in ("s", "es", "ed", "d", "ing", "ly", "er", "ers", "ies", "al", "ity", "ment", "ments", "ness"):
        if w.endswith(suf) and len(w) - len(suf) >= 3:
            stem = w[: -len(suf)]
            if stem in _DICT or stem + "e" in _DICT or stem + "y" in _DICT:
                return True
    return False


def clean_pdf_cut(cut, full):
    """Turn a pdftotext abstract cut into a plain paragraph.

    Line-broken hyphens are rejoined only when the joined word occurs elsewhere
    in the PDF text and the hyphenated form does not; otherwise the hyphen stays.
    """
    if not cut:
        return None
    for k, v in LIGATURES.items():
        cut = cut.replace(k, v)
    words_full = full.lower()

    def join(m):
        a, b = m.group(1), m.group(2)
        joined, hyph = (a + b).lower(), (a + "-" + b).lower()
        jw = len(re.findall(r"\b" + re.escape(joined) + r"\b", words_full))
        hw = len(re.findall(r"\b" + re.escape(hyph) + r"\b", words_full))
        if hw > 0 and jw == 0:
            return a + "-" + b
        if jw > 0:
            return a + b
        if dict_word(joined):
            return a + b
        if dict_word(a) and dict_word(b):
            return a + "-" + b
        return a + b if b[:1].islower() else a + "-" + b

    cut = re.sub(r"(\w+)-\s*\n\s*(\w+)", join, cut)
    cut = cut.replace("\u00ad", "")
    cut = re.sub(r"[-\u2010\u2011]{2,}", "-", cut)
    cut = cut.replace("\u2018\u2018", "\u201c").replace("\u2019\u2019", "\u201d")
    cut = re.sub(r"\s+", " ", cut).strip()

    def split_joined(m):
        # "contextaware" -> "context-aware" when the PDF itself spells it that way elsewhere
        w = m.group(0)
        if dict_word(w):
            return w
        for i in range(2, len(w) - 1):
            a, b = w[:i], w[i:]
            if re.search(r"\b" + re.escape(a + "-" + b) + r"\b", full, re.I):
                return a + "-" + b
        return w

    cut = re.sub(r"\b[a-z]{5,}\b", split_joined, cut)
    return strip_boilerplate(cut)


BOILER_TAIL = re.compile(
    r"\s*(?:\u00a9|\(c\)\s*\d{4}|\u00d3\s*\d{4}|[\x02\x03]\s*\d{4}|Copyright\b|Received:?\s+(?:\d|[A-Z][a-z]+ \d)"
    r"|Article history|All rights reserved).*$",
    re.S,
)


def strip_boilerplate(t):
    """Drop a trailing publisher copyright or received/accepted block."""
    m = BOILER_TAIL.search(t)
    if m and m.start() > 0.6 * len(t) and len(re.findall(r"\w+", t[: m.start()])) >= 40:
        return t[: m.start()].rstrip()
    return t

def hyphen_artifacts(t, full):

    out = []

def gaps(a, ref, n=5):
    """Runs of ref's tokens not covered by any n-gram shared with a.
    Returns [(start, length, 'start'|'mid'|'end', preview)]."""
    rt, ag = normalize(ref), grams(normalize(a), n)
    cov = [False] * len(rt)
    for i in range(len(rt) - n + 1):
        if tuple(rt[i : i + n]) in ag:
            for j in range(i, i + n):
                cov[j] = True
    runs, i = [], 0
    while i < len(rt):
        if cov[i]:
            i += 1
            continue
        j = i
        while j < len(rt) and not cov[j]:
            j += 1
        where = "start" if i == 0 else ("end" if j == len(rt) else "mid")
        runs.append((i, j - i, where, " ".join(rt[i:j][:10])))
        i = j
    return runs


def continuation_ok(bib_toks, full_toks):
    """True if, in the PDF text, the bib abstract is followed by a heading-like token
    (no truncation), or if we cannot locate it."""
    if len(bib_toks) < 10 or not full_toks:
        return None
    tail = tuple(bib_toks[-5:])
    n = len(full_toks)
    hits = [i for i in range(n - 4) if tuple(full_toks[i : i + 5]) == tail]
    if not hits:
        return None
    head = tuple(bib_toks[:5])
    starts = [i for i in range(n - 4) if tuple(full_toks[i : i + 5]) == head]
    if starts:
        hits = [h for h in hits if h >= starts[0]] or hits
    i = hits[0] + 5
    nxt = full_toks[i : i + 3]
    return bool(nxt) and nxt[0] in END_WORDS

        if os.path.exists(p):
            r = json.load(open(p, encoding="utf-8"))
            s = {"status": r.get("status"), "pages": r.get("pages")}
            if r.get("status") == 200:
                full_text = r.get("full") or ""
                s["usable"] = usable_text(full_text)
                s["chars"] = len(full_text)
                s["title_in_head"] = round(
                    containment(set(normalize(e.get("title") or "")), set(normalize((r.get("head") or "")[:4000]))), 2
                )
                s["ocr"] = r.get("ocr", False)
                cands = [(t, h) for t, h in pdf_cuts(r.get("head") or "")]
                cands += [(t, h + "(wide)") for t, h in pdf_cuts(r.get("wide") or "")]
                cands = [(clean_pdf_cut(t, full_text), h) for t, h in cands]
                good = [c for c in cands if trustworthy_pdf_cut(c[0]) and "+prose" not in c[1]]
                if good or cands:
                    cut, how = (good or cands)[0]
                    s["cut_how"] = how
                    s["cut_words"] = len(normalize(cut))
                    s["cut_trust"] = trustworthy_pdf_cut(cut)
                    refs["pdf_cut"] = (cut, s)
                s["has_abstract_word"] = bool(
                    re.search(r"(?im)^[ \t\f]*(?:a\s?b\s?s\s?t\s?r\s?a\s?c\s?t|summary)\b", (r.get("wide") or ""))
                )
            rec["sources"]["pdf"] = s

    full_toks = normalize(full_text) if full_text else []
    full_g = grams(full_toks)
    pdf_usable = bool(full_text) and usable_text(full_text)

    # forward scores: bib abstract inside each reference
    fwd = {}
    for name, (txt, s) in refs.items():
        if name == "pdf_cut":
            continue
        fwd[name] = containment(bg, grams(normalize(txt)))
    if full_text:
        fwd["pdf_full"] = containment(bg, full_g)
    if "pdf_cut" in refs:
        fwd["pdf_cut"] = containment(bg, grams(normalize(refs["pdf_cut"][0])))
    rec["fwd"] = {k: round(v, 3) for k, v in fwd.items()}

    # reverse scores: each reference abstract covered by the bib abstract (completeness)
    rev = {}
    agree = {}
    for name, (txt, s) in refs.items():
        rg = grams(normalize(txt))
        rev[name] = containment(rg, bg)
        if pdf_usable and name != "pdf_cut":
            agree[name] = containment(rg, full_g)
    rec["rev"] = {k: round(v, 3) for k, v in rev.items()}
    rec["api_in_pdf"] = {k: round(v, 3) for k, v in agree.items()}

    # alignment against the PDF text tolerates extraction damage but not edits
    align = align_ratio(bib_abs, full_text) if pdf_usable else 0.0
    rec["align_pdf"] = round(align, 3)

    best = max(fwd.values()) if fwd else 0.0
    rec["best"] = round(best, 3)
    rec["best_src"] = max(fwd, key=fwd.get) if fwd else ""

    pdf_s = rec["sources"].get("pdf", {})
    cut_trusted = "pdf_cut" in refs and pdf_s.get("cut_trust") and pdf_s.get("title_in_head", 0) >= 0.8

    def api_ok(name):
        """An API abstract is used only if it is the right work, clean, and complete."""
        txt, s = refs[name]
        problems = []
        if s.get("title_sim", 1.0) < T_TITLE:
            problems.append("title")
        if not trustworthy_api(txt):
            problems.append("junk-or-length")
        if full_text:
            arts = hyphen_artifacts(txt, full_text)
            if arts:
                problems.append("hyphenation:" + ",".join(arts[:3]))
        if pdf_usable:
            if agree.get(name, 0) < T_AGREE:
                problems.append(f"not-in-pdf:{agree.get(name, 0):.2f}")
            xg = grams(normalize(txt))
            if cut_trusted:
                cg = grams(normalize(refs["pdf_cut"][0]))
                x_in_cut, cut_in_x = containment(xg, cg), containment(cg, xg)
                if x_in_cut < T_REV:
                    problems.append(f"beyond-pdf-abstract:{x_in_cut:.2f}")
                complete = cut_in_x >= T_REV or continuation_ok(normalize(txt), full_toks)
            else:
                complete = ends_cleanly(txt)
            if not complete:
                problems.append("maybe-truncated")
        elif not ends_cleanly(txt):
            problems.append("maybe-truncated")
        s["problems"] = problems
        return not problems

    api_pref = [n for n in ("crossref", "arxiv", "openalex") if n in refs]
    good_api = [n for n in api_pref if api_ok(n)]
    for n in api_pref:
        if n not in good_api:
            rec["flags"].append(f"untrusted:{n}")

    # the replacement we would use: trusted API abstract first, then a clean PDF cut
    cand = None
    if good_api:
        cand = good_api[0]
    elif cut_trusted and pdf_usable:
        cand = "pdf_cut"
    if cand:
        rec["candidate_src"], rec["candidate"] = cand, refs[cand][0]

    verbatim_api = [n for n in api_pref if fwd[n] >= T_HI]
    verbatim_pdf = pdf_usable and (fwd.get("pdf_full", 0) >= T_HI or align >= T_ALIGN)

    action, why, use = None, "", None
    if verbatim_api:
        comp = max(rev[n] for n in verbatim_api)
        trusted = [n for n in good_api if n in verbatim_api]
        if trusted:
            # complete means no run of T_GAP+ words of a verified abstract is missing
            missing = {n: [g for g in gaps(bib_abs, refs[n][0]) if g[1] >= T_GAP] for n in trusted}
            rec["missing"] = {n: [f"{g[2]}:{g[3]}" for g in v] for n, v in missing.items() if v}
            whole = [n for n in trusted if not missing[n]]
            cut_whole = (
                cut_trusted
                and fwd.get("pdf_cut", 0) >= T_HI
                and not [g for g in gaps(bib_abs, refs["pdf_cut"][0]) if g[1] >= T_GAP and g[2] != "start"]
            )
            if whole:
                action, why = "VERBATIM", f"in {','.join(verbatim_api)}; nothing missing from {whole[0]}"
            elif cut_whole:
                action, why = "VERBATIM", f"nothing missing from the PDF abstract; {trusted[0]} has another version"
            else:
                n = trusted[0]
                where = ",".join(sorted({g[2] for g in missing[n]}))
                action, why, use = "REPLACE", f"incomplete: {where} of {n} missing", n
        elif comp >= T_REV:
            action, why = "VERBATIM", f"in untrusted {','.join(verbatim_api)}; covers {comp:.2f} of it"
        else:
            action, why = "REVIEW", f"in untrusted {','.join(verbatim_api)}; covers {comp:.2f} of it"
    elif verbatim_pdf:
        cut_ok = (
            "pdf_cut" in refs
            and fwd.get("pdf_cut", 0) >= T_HI
            and not [g for g in gaps(bib_abs, refs["pdf_cut"][0]) if g[1] >= T_GAP and g[2] != "start"]
        )
        cont = continuation_ok(bib_toks, full_toks)
        rec["continues_to_heading"] = cont
        differs = f"; {good_api[0]} has another version" if good_api else ""
        # a cut-short copy: the bib text sits inside a clean PDF abstract that goes on
        cut_missing = []
        if cut_trusted and fwd.get("pdf_cut", 0) >= T_HI:
            cut_missing = [g for g in gaps(bib_abs, refs["pdf_cut"][0]) if g[1] >= T_GAP and g[2] != "start"]
        if cut_ok:
            action, why = "VERBATIM", "in PDF; covers the PDF abstract cut" + differs
        elif cont:
            action, why = "VERBATIM", "in PDF; followed by a heading" + differs
        elif cut_missing:
            where = ",".join(sorted({g[2] for g in cut_missing}))
            action, why, use = "REPLACE", f"incomplete: {where} of the PDF abstract missing", "pdf_cut"
        else:
            action, why = "REVIEW", "in PDF; completeness unclear" + differs
    elif best < T_LO:
        if cand:
            action, why, use = "REPLACE", f"paraphrase; {cand} abstract", cand
        elif api_pref:
            action, why = "REVIEW", "paraphrase; API abstract untrusted"
        elif pdf_usable and not pdf_s.get("has_abstract_word"):
            if pdf_s.get("title_in_head", 0) >= 0.5:
                action, why = "DELETE", "paraphrase; source has no abstract"
            else:
                action, why = "REVIEW", "paraphrase; PDF title mismatch"
        elif pdf_usable:
            action, why = "REVIEW", "paraphrase; PDF has an Abstract heading but no clean cut"
        else:
            action, why = "REVIEW", "paraphrase?; no usable reference text"
    elif good_api:
        # a clean, verified abstract exists and the bib fails both T_HI against it and
        # T_ALIGN against the PDF: an edited or condensed copy, not extraction noise
        action, why, use = "REPLACE", f"edited (best {best:.2f}); {good_api[0]} abstract", good_api[0]
    else:
        action, why = "REVIEW", f"middle score {best:.2f}"
    replacement = (refs[use][0], use) if use else None

    rec["action"], rec["why"] = action, why
    if replacement:
        rec["replacement"], rec["replacement_src"] = replacement
    return rec


def abstract_sha(value):
    import hashlib

    return hashlib.sha1(re.sub(r"\s+", " ", value).strip().encode("utf-8")).hexdigest()

def bib_escape(s):
    """Make plain text safe inside a braced BibTeX value without changing its words."""
    s = re.sub(r"\s+", " ", s).strip()
    # keep braces balanced: drop any unmatched ones
    out, depth = [], 0
    for c in s:
        if c == "{":
            depth += 1
        elif c == "}":
            if depth == 0:
                continue
            depth -= 1
        out.append(c)
    s = "".join(out)
    while depth > 0:
        i = s.rfind("{")
        s = s[:i] + s[i + 1 :]
        depth -= 1
    return s


def source_text(e, spec):
    """Text named by an override: {"from": "crossref"|"openalex"|"arxiv"|"cut:N"|"span",
    "start": ..., "end": ..., "edits": [[old, new], ...]}. "span" takes the PDF text
    from the first occurrence of `start` through the following `end`, inclusive,
    cleaned like a cut. Each edit must match exactly once."""
    src = spec["from"]
    doi = entry_doi(e)[0]
    if src == "crossref":
        t = fetch_crossref(doi)["abstract"]
    elif src == "openalex":
        t = fetch_openalex(doi)["abstract"]
    elif src == "arxiv":
        with open(cache_path("arxiv", arxiv_id(e)), encoding="utf-8") as f:
            t = json.load(f)["abstract"]
    else:
        with open(cache_path("pdf", e.key), encoding="utf-8") as f:
            d = json.load(f)
        if src.startswith("cut:"):
            cands = pdf_cuts(d.get("head") or "") + pdf_cuts(d.get("wide") or "")
            t = clean_pdf_cut(cands[int(src[4:])][0], d["full"])
        elif src == "span":
            full = d["full"]
            for k, v in LIGATURES.items():
                full = full.replace(k, v)
            loose = lambda x: re.compile(r"\s+".join(map(re.escape, x.split())))
            ms = loose(spec["start"]).search(full)
            me = ms and loose(spec["end"]).search(full, ms.start())
            if not (ms and me) or me.end() - ms.start() > 6000:
                raise SystemExit(f"span anchors not found (or span too long) in {e.key}")
            t = clean_pdf_cut(full[ms.start() : me.end()], full)
        else:
            raise SystemExit(f"unknown source {src} for {e.key}")
    if spec.get("drop_from"):
        # drop a trailing non-abstract block (funding, supplement, license lines)
        if t.count(spec["drop_from"]) != 1:
            raise SystemExit(f"drop_from marker matches {t.count(spec['drop_from'])} times in {e.key}")
        t = t[: t.index(spec["drop_from"])].rstrip()
    for old, new in spec.get("edits", []):
        if t.count(old) != 1:
            raise SystemExit(f"edit {old!r} matches {t.count(old)} times in {e.key}")
        t = t.replace(old, new)
    return t


def cmd_apply(bib, overrides=None, dry=False, keys=None):
    text, entries = load(bib)
    with open(REPORT, encoding="utf-8") as f:
        recs = {r["key"]: r for r in json.load(f)}
    manual = {}
    if overrides:
        with open(overrides, encoding="utf-8") as f:
            manual = json.load(f)  # key -> {"action": "KEEP"|"DELETE"|"REPLACE", "from": ..., "note": ...}
    edits = []  # (start, end, new)
    summary, unresolved = {}, []
    for e in entries:
        f = e.fields.get("abstract")
        if not f:
            continue
        if keys and e.key not in keys:
            continue
        r = recs.get(e.key)
        if r is None:
            if keys:
                raise SystemExit(f"{e.key} is not in the report; rerun score with the same --keys")
            continue  # the report covers only some entries (score --keys)
        if r["bib_head"] != re.sub(r"\s+", " ", f.value)[:120]:
            raise SystemExit(f"report is stale for {e.key}; rerun score")
        if e.key in manual:
            m = manual[e.key]
            act = m["action"]
            new = source_text(e, m) if act == "REPLACE" else None
            tag = act + "(manual)"
        elif r["action"] in ("REPLACE", "DELETE"):
            act, new, tag = r["action"], r.get("replacement"), r["action"]
        elif r["action"] == "REVIEW":
            unresolved.append(e.key)
            continue
        else:
            continue
        if act == "DELETE":
            edits.append((f.line_start, f.line_end, ""))
        elif act == "REPLACE":
            edits.append((f.vstart, f.vend, bib_escape(new)))
        elif act != "KEEP":
            raise SystemExit(f"bad action {act} for {e.key}")
        summary[tag] = summary.get(tag, 0) + 1
    if unresolved:
        raise SystemExit(f"{len(unresolved)} REVIEW entries without an override: {unresolved}")
    for s_, t_, new in sorted(edits, reverse=True):
        text = text[:s_] + new + text[t_:]
    print(json.dumps(summary, indent=1))
    if not dry:
        with open(bib, "w", encoding="utf-8") as fh:
            fh.write(text)

# citation-hunt

Find citations for all claims in a section of text - both uncited and already-cited ones. Works in three chapters:

- **Chapter 1 - Bib scan:** Decompose the section into paragraph-level claims, then scan every entry in `references.bib` using titles, abstracts, and `annote` fields to find the best match for each claim.
- **Chapter 2 - Upstream references:** For every paper matched in Chapter 1, read its reference list and trace one level upstream to find additional supporting sources not already in the bib.
- **Chapter 3 - World knowledge + web:** For claims still uncovered or only weakly covered, use world knowledge and targeted web searches to find papers that should exist.

## Invocation

```
/citation-hunt [section] [--file path/to/file.org]
```

- `section`: a heading name (partial match OK) or line range like `100-120`
- `--file`: path to the org/tex file to search. If omitted, defaults to `~/org/thesis/thesis.org`

The `--seeds` argument has been removed. Chapter 1 scans the entire bib. Chapter 2 traces upstream from whatever Chapter 1 finds.

---

## Phase 0: Write scratch file

Before doing anything else, establish a scratch directory:

```bash
mkdir -p ~/.cache/citation-hunt/scratch
```

File path: `~/.cache/citation-hunt/scratch/<section-slug>.org`

The slug is the heading text lowercased, spaces replaced with hyphens, non-alphanumeric stripped (e.g., "Problem Statement" -> `problem-statement`).

<verbatim paragraph text>

,** Para 2 (lines <C>-<D>)
<verbatim paragraph text>

   - [ ] @citekey (conf NN) - one line reason
   - [ ] @citekey2 (conf NN) - one line reason

Mark [ ] -> [X] as each paper is fully processed (roam + thesis.org both done).
,**This list is the authoritative work queue for Phase 3.5.** Read it at the start
of each paper's processing cycle rather than reconstructing from context.

(written incrementally during Chapter 1 - leave blank until then)

Delete this file at the end of Phase 6.

## Phase 1: Extract and confirm target text

### 1a. Locate the file

If `--file` was given, use that path. Otherwise use `~/org/thesis/thesis.org`.

### 1b. Extract the section

Read the file. Find the section matching the given heading (case-insensitive partial match) or line range.

Extract:
- The heading text
- All body text under that heading, stopping at the next same-level or higher heading
- Any existing cite keys already present (org `[[cite:@key]]` or LaTeX `\cite{key}` format)

,**Segment by paragraph:** Split the extracted body text at blank lines. Each non-empty block is a paragraph. Record the line range for each paragraph.

Write the scratch file now (Phase 0 format above, CLAIMS and MATCH STATE sections left blank).

### 1c. Confirm with user

## Phase 2: Decompose into claims

Read the scratch file. For each paragraph, break it into individual claims. A claim is any sentence or clause that:
- States a fact about the world ("software systems degrade over time")
- Makes a motivation or justification ("this is problematic because...")
- States a prevalence or frequency ("decisions are rarely documented")
- Attributes a property to a concept ("architecture erosion causes...")

Skip: purely definitional sentences that introduce your own framework/terminology (those don't need external citations), and transition sentences.

,**Do NOT skip already-cited sentences.** Process every claim regardless of whether it already has a citation - the goal is the *best* citation, not just any citation.

For each claim, record:
- **Para**: which paragraph it came from (number and line range)
- **Type**: one of:
  - `count/prevalence` - asserts how often, how many, or how common something is. Needs a measured quantity, survey count, or documented frequency. Example: "architects make many decisions", "documentation is rarely maintained"
  - `causal/consequence` - asserts that X causes Y or that a phenomenon produces a specific outcome. Needs a mechanism, longitudinal study, or controlled evidence. Example: "knowledge loss increases maintenance costs"
  - `canonical/definitional` - introduces an established concept or framework. Needs the paper that coined or canonically established it. Example: "bounded rationality", "architecture erosion", "residual causality"
  - `cognitive/behavioral` - claims about how people think, decide, or behave. Needs psychology, HCI, or cognitive science study. Example: "experts rely on pattern recognition", "decision fatigue impairs judgment"
  - `observation-in-practice` - asserts that something happens in real software projects. Needs a case study, interview study, or empirical observation in actual projects. Example: "ADRs are rarely written", "architects face time pressure"
  - `design-rationale` - justifies a specific approach or tool. Needs prior work proposing or evaluating that approach. Example: "ADRs reduce knowledge vaporization"

Also record for each claim: the **minimum evidence bar** - one sentence describing what a PhD examiner would require to accept a citation for this claim. Be specific. Example: "a study that counted or surveyed the number of architectural decisions made per project" - not just "a paper about architectural decisions."

Write the CLAIMS section of the scratch file:

```org

,** Para 1 (lines <A>-<B>)
- [count/prevalence | bar: study measuring decision frequency or volume per project] architects make many decisions
- [causal/consequence | bar: empirical evidence linking undocumented decisions to maintenance cost increase] knowledge loss increases costs

,** Para 2 (lines <C>-<D>)
- [canonical/definitional | bar: paper that coined or canonically defined the concept] architecture erosion is divergence between intended and actual structure
...
```

## Phase 3: Match all claims against the full bibliography

### 3a. Load bib

Read `~/org/bib/references.bib`. For each entry, collect:
- citekey
- title
- abstract (if present)
- `annote` field (concise summary - use this as the primary matching signal when present)
- keywords (if present)
- `file` field (whether a local PDF exists)

### 3b. Ideal source reflection

Before scanning the bib, for each claim write a one-paragraph ideal source description. Answer:
- What methodology would this paper use? (e.g. controlled study, survey of N practitioners, systematic literature review, longitudinal case study, RCT, theoretical framework with empirical validation)
- What would its key finding or contribution look like? (e.g. "found that architects make on average X decisions per project", "defined and operationalized the concept of Y", "showed through interviews that Z is common in practice")
- What would disqualify a paper from satisfying this claim? (e.g. "a paper that only argues for X without measuring it", "a position paper", "a paper about a related but distinct phenomenon")

This description is your matching target for the bib scan - not the claim text itself.

Example for claim "architects make many decisions" (type: count/prevalence):
> Ideal: an empirical study - observational, survey, or case study - that measured or documented the volume of architectural decisions made during a software project. Key finding would be a count, frequency, or documented pattern of decision-making activity. Disqualifies: any paper that asserts decision volume without measuring it, position papers arguing for decision-centric architecture, conceptual frameworks.

### 3c. Match every claim against every bib entry

Process claims **one paragraph at a time**. After finishing all claims in a paragraph, immediately append those rows to the `* MATCH STATE` section of the scratch file before moving to the next paragraph. This ensures compaction cannot wipe in-progress work.

,**Resuming after compaction:** If context compaction occurs mid-scan, re-read the scratch file at the start of the next message. Find the last paragraph recorded in `* MATCH STATE`. Resume scoring from the *next* paragraph - do not re-score paragraphs already written to MATCH STATE. Any paragraph whose rows are not yet in MATCH STATE must be re-scored from scratch (its in-progress work was lost).

For each claim, scan all bib entries and score them against the **ideal source description** from 3b - not the claim text. Use the `annote` as the primary signal. Fall back to abstract and title if `annote` is absent.

Scoring guidance:
- **direct (80-100):** The `annote` or abstract explicitly states the same phenomenon the claim asserts. The paper is clearly about this.
- **indirect (40-79):** The `annote`/abstract covers a related concept that supports the claim but is not the central point.
- **weak (10-39):** Tangential overlap - the paper is in the area but the claim is peripheral to what it studies.
- **none (0-9):** No meaningful overlap.

Hard disqualification rules by claim type - apply these before scoring. A paper that fails a disqualification rule is capped at score 39 regardless of topical overlap:
- `count/prevalence`: disqualify papers whose annote/abstract contains none of: "found that", "measured", "survey of", "interviewed", "N projects", "N architects", "empirical", "case study", "% of", "on average". Position papers, opinion pieces, and conceptual frameworks are disqualified.
- `causal/consequence`: disqualify papers that only assert the causal relationship without providing a mechanism, longitudinal evidence, or controlled study.
- `canonical/definitional`: survey and foundational papers are explicitly preferred. Do NOT disqualify.
- `cognitive/behavioral`: disqualify papers from SE venues that only discuss the phenomenon without grounding in psychology, cognitive science, or HCI study methodology.
- `observation-in-practice`: disqualify papers that are purely theoretical or that study student/toy projects rather than real-world software development.
- `design-rationale`: prefer papers that evaluate or propose the approach empirically; deprioritize papers that merely mention it.

For each claim, select the best candidate(s). A claim may have multiple good matches - list up to 3, ranked by score. Only list a candidate if score >= 40. Weak matches (10-39) are noted but not recommended.

### 3d. Examiner judgment

For each claim-candidate pair with score >= 40, apply an explicit PhD examiner judgment. Ask: *"If a PhD examiner saw this citation for this claim, would they accept it as genuine evidence, or would they push back?"*

Classify each pair as one of:

- **`DIRECT_EVIDENCE`**: the paper directly provides the kind of evidence the claim requires (matches the minimum evidence bar from Phase 2). A passage in this paper will satisfy a PhD examiner. Proceed to Phase 3.5.
- **`UPSTREAM_NEEDED`**: the paper is topically relevant and likely cites the actual primary evidence, but is itself a position paper, framing paper, survey, or opinion piece that doesn't directly evidence the claim. The real value is in its reference list. **Skip Phase 3.5 for this paper. Add to Chapter 2 seed list immediately.**
- **`REJECTED`**: on reflection, this paper doesn't actually support this claim even indirectly. Score downgraded to < 40. Move to uncovered.

The key distinction between `DIRECT_EVIDENCE` and `UPSTREAM_NEEDED`: ask "if I find a passage in this paper, will it contain a measurement, a documented observation, or an established definition - or will it just restate the same claim without grounding?" If the latter, it is `UPSTREAM_NEEDED`, not `DIRECT_EVIDENCE`.

,**After completing each paragraph's claims:** append the new rows to the `* MATCH STATE` section of the scratch file before moving to the next paragraph. Do not wait until all paragraphs are done.

- **Basis**: one sentence explaining the judgment

#!/usr/bin/env python3
"""
Find the bounding box of a quote on a PDF page and write a highlight annotation.

Usage:
    python3 bbox.py --pdf <path> --page <N> --quote "<text>" [--color green]

Output (on success, to stdout):
    NOTER_PAGE: (N x1 . y1)
    HIGHLIGHT: #s(pdf-highlight N ((y1 x1 y2 x2)))

Also writes a highlight annotation directly into the PDF at the matched word spans.
Color is an RGB tuple name: green (citation-hunt), yellow (default manual).

On MATCH_FAILED: exits with code 1 and prints reason to stderr.

Extraction fallback ladder:
  1. pdftotext -bbox-layout  -> precise word bboxes via XHTML
  2. XML sanitize + retry     -> handles control chars / unescaped & in text layer
  3. pypdf visitor API        -> char-level positions, zero extra deps
  4. pdfminer.six             -> best encoding coverage, handles more PDF variants
  5. pdftoppm visual render   -> page rendered to PNG, coords estimated from image read
  6. MATCH_FAILED             -> all extraction methods exhausted
"""

import argparse
import os
import re
import subprocess
import sys
import tempfile
import xml.etree.ElementTree as ET

COLORS = {
    'green':  (0, 1, 0),
    'yellow': (1, 1, 0),
    'cyan':   (0, 1, 1),
}


# ---------------------------------------------------------------------------
# Normalization (shared by all tiers)
# ---------------------------------------------------------------------------

def normalize(s):
    s = re.sub(r'-', ' ', s.lower())
    return re.sub(r'[^\w\s]', '', s).split()


# ---------------------------------------------------------------------------
# Span matching (shared by all tiers)
# words: list of (text, x1, y1_top, x2, y2_top) in top-left origin coords
# Returns matched word sublist or raises ValueError.
# ---------------------------------------------------------------------------

def match_span(words, quote):
    q_words = normalize(quote)
    if len(q_words) < 4:
        raise ValueError("Quote too short for reliable matching (< 4 words)")

    anchor_start = q_words[:4]
    anchor_end   = q_words[-4:]

    flat = []  # (norm_word, word_idx)
    for wi, w in enumerate(words):
        for nw in normalize(w[0]):
            flat.append((nw, wi))

    if len(flat) < 8:
        raise ValueError("Too few words on page for matching")

    start_flat = None
    for i in range(len(flat) - 4):
        span = [flat[i + j][0] for j in range(4)]
        if sum(a == b for a, b in zip(span, anchor_start)) >= 3:
            start_flat = i
            break
    if start_flat is None:
        raise ValueError("Start anchor not found")

    end_flat = None
    search_limit = min(start_flat + len(q_words) + 80, len(flat) - 4)
    for i in range(start_flat, search_limit):
        span = [flat[i + j][0] for j in range(4)]
        if sum(a == b for a, b in zip(span, anchor_end)) >= 3:
            end_flat = i + 3
            break
    if end_flat is None:
        raise ValueError("End anchor not found")

    start_wi = flat[start_flat][1]
    end_wi   = flat[end_flat][1]
    return words[start_wi:end_wi + 1]


# ---------------------------------------------------------------------------
# Tier 1: pdftotext -bbox-layout
# ---------------------------------------------------------------------------

def _sanitize_xml(raw):
    """Strip control characters and fix bare ampersands that break XML parsing."""
    cleaned = re.sub(r'[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]', '', raw)
    cleaned = re.sub(r'&(?!(?:[a-zA-Z][a-zA-Z0-9]*|#[0-9]+|#x[0-9a-fA-F]+);)', '&amp;', cleaned)
    return cleaned


def tier1_pdftotext(pdf_path, page):
    """Returns (words, W, H) with top-left origin coords, or raises RuntimeError."""
    result = subprocess.run(
        ['pdftotext', '-f', str(page), '-l', str(page), '-bbox-layout', pdf_path, '-'],
        capture_output=True, text=True
    )
    if result.returncode != 0 and not result.stdout.strip():
        raise RuntimeError(f"pdftotext failed: {result.stderr.strip()}")

    raw = result.stdout

    # Try parse as-is, then with sanitization
    root = None
    for attempt, xml_str in enumerate([raw, _sanitize_xml(raw)], 1):
        try:
            root = ET.fromstring(xml_str)
            break
        except ET.ParseError as e:
            if attempt == 2:
                raise RuntimeError(f"XML parse failed after sanitization: {e}")

    NS = 'http://www.w3.org/1999/xhtml'
    page_el = root.find(f'.//{{{NS}}}page')
    if page_el is None:
        raise RuntimeError("No <page> element in pdftotext bbox output")

    W = float(page_el.get('width'))
    H = float(page_el.get('height'))
    words = [
        (w.text or '', float(w.get('xMin')), float(w.get('yMin')),
         float(w.get('xMax')), float(w.get('yMax')))
        for w in root.iter(f'{{{NS}}}word')
    ]
    if not words:
        raise RuntimeError("pdftotext returned zero words")
    return words, W, H


# ---------------------------------------------------------------------------
# Tier 3: pypdf visitor API
# coords from pypdf are PDF-native (bottom-left origin); we flip to top-left.
# ---------------------------------------------------------------------------

def tier3_pypdf(pdf_path, page):
    """Returns (words, W, H) with top-left origin coords, or raises RuntimeError."""
    from pypdf import PdfReader

    reader = PdfReader(pdf_path)
    if page < 1 or page > len(reader.pages):
        raise RuntimeError(f"Page {page} out of range (doc has {len(reader.pages)} pages)")

    page_obj = reader.pages[page - 1]
    mb = page_obj.mediabox
    W = float(mb.width)
    H = float(mb.height)

    # Visitor yields (text, cm, tm, fontname, fontsize)
    # tm[4]=x, tm[5]=y in PDF coords (bottom-left origin)
    chunks = []  # (text, x_bl, y_bl, fontsize)

    def visitor(text, cm, tm, font, size):
        if text and text.strip():
            x, y = float(tm[4]), float(tm[5])
            sz = float(size) if size else 10.0
            chunks.append((text, x, y, sz))

    page_obj.extract_text(visitor_text=visitor)

    if not chunks:
        raise RuntimeError("pypdf visitor yielded no text chunks")

    # Split chunks into individual words; estimate bbox from font size
    words = []
    for (text, x_bl, y_bl, sz) in chunks:
        # Split on whitespace, estimate each word's x offset
        # pypdf gives us the start-of-chunk x; advance crudely by char_width ~ 0.5*sz
        char_w = sz * 0.5
        token_x = x_bl
        for token in re.split(r'(\s+)', text):
            if not token.strip():
                token_x += len(token) * char_w * 0.4
                continue
            w_width = len(token) * char_w
            x1 = token_x
            x2 = token_x + w_width
            # PDF y_bl is baseline (bottom-left); top of glyph ~ y_bl + sz*0.8
            y_top_pdf = y_bl + sz * 0.8
            y_bot_pdf = y_bl
            # Flip to top-left origin for match_span / write_highlight consistency
            y1_top = H - y_top_pdf
            y2_top = H - y_bot_pdf
            words.append((token, x1, y1_top, x2, y2_top))
            token_x += w_width

    if not words:
        raise RuntimeError("pypdf visitor: no word tokens after splitting")
    return words, W, H


# ---------------------------------------------------------------------------
# Tier 4: pdfminer.six
# ---------------------------------------------------------------------------

def tier4_pdfminer(pdf_path, page):
    """Returns (words, W, H) with top-left origin coords, or raises RuntimeError."""
    from pdfminer.high_level import extract_pages
    from pdfminer.layout import LTTextBox, LTTextLine, LTChar, LTAnno, LTPage

    words = []
    W, H = None, None

    for page_num, page_layout in enumerate(extract_pages(pdf_path), 1):
        if page_num != page:
            continue
        W = page_layout.width
        H = page_layout.height

        for element in page_layout:
            if not isinstance(element, LTTextBox):
                continue
            for line in element:
                if not isinstance(line, LTTextLine):
                    continue
                # Accumulate characters into word tokens
                current_word = []
                current_chars = []  # (x0, y0, x1, y1) per char in pdfminer coords (bottom-left)

                def flush_word():
                    if not current_word:
                        return
                    text = ''.join(current_word)
                    if text.strip():
                        x1 = min(c[0] for c in current_chars)
                        y1_pdf = min(c[1] for c in current_chars)
                        x2 = max(c[2] for c in current_chars)
                        y2_pdf = max(c[3] for c in current_chars)
                        # pdfminer uses bottom-left origin; flip to top-left
                        y1_top = H - y2_pdf
                        y2_top = H - y1_pdf
                        words.append((text, x1, y1_top, x2, y2_top))
                    current_word.clear()
                    current_chars.clear()

                for char in line:
                    if isinstance(char, LTChar):
                        if char.get_text() in (' ', '\t', '\n'):
                            flush_word()
                        else:
                            current_word.append(char.get_text())
                            current_chars.append((char.x0, char.y0, char.x1, char.y1))
                    elif isinstance(char, LTAnno):
                        if char.get_text() in (' ', '\t', '\n'):
                            flush_word()
                flush_word()
        break

    if W is None:
        raise RuntimeError(f"pdfminer: page {page} not found in document")
    if not words:
        raise RuntimeError("pdfminer: no words extracted")
    return words, W, H


# ---------------------------------------------------------------------------
# Tier 5: visual render via pdftoppm
# Renders page to PNG, reads it with the Read tool (caller handles this).
# Returns estimated coords from visual inspection - we do our best here.
# Since this runs as a script, we emit a special VISUAL_FALLBACK line and
# let the caller (citation-hunt skill) handle the image read step.
# ---------------------------------------------------------------------------

def tier5_visual_render(pdf_path, page):
    """
    Render page to PNG and print VISUAL_FALLBACK with the image path.
    The caller (citation-hunt) reads the image, visually locates the passage,
    and estimates fractional coords manually.
    Raises RuntimeError if pdftoppm is unavailable.
    """
    pdftoppm = subprocess.run(['which', 'pdftoppm'], capture_output=True, text=True)
    if pdftoppm.returncode != 0:
        raise RuntimeError("pdftoppm not available for visual fallback")

    tmp_dir = tempfile.mkdtemp(prefix='bbox_visual_')
    out_prefix = os.path.join(tmp_dir, 'page')
    result = subprocess.run(
        ['pdftoppm', '-f', str(page), '-l', str(page), '-r', '150', '-png', pdf_path, out_prefix],
        capture_output=True, text=True
    )
    if result.returncode != 0:
        raise RuntimeError(f"pdftoppm failed: {result.stderr.strip()}")

    # Find the generated PNG
    pngs = sorted(f for f in os.listdir(tmp_dir) if f.endswith('.png'))
    if not pngs:
        raise RuntimeError("pdftoppm produced no PNG output")

    img_path = os.path.join(tmp_dir, pngs[0])
    # Signal to caller: they should read this image and estimate coords
    print(f"VISUAL_FALLBACK: {img_path}", file=sys.stderr)
    raise RuntimeError(f"VISUAL_FALLBACK:{img_path}")


# ---------------------------------------------------------------------------
# Write highlight (unchanged from original; coords are top-left origin)
# ---------------------------------------------------------------------------

def write_highlight(pdf_path, page, word_span, W, H, color_rgb):
    """Write highlight annotation into PDF using per-word QuadPoints."""
    from pypdf import PdfReader, PdfWriter
    from pypdf.generic import (
        ArrayObject, DictionaryObject, FloatObject, NameObject,
        NumberObject, ByteStringObject,
    )
    import random, string

    reader = PdfReader(pdf_path)
    writer = PdfWriter()
    writer.append(reader)

    page_obj = writer.pages[page - 1]
    media_box = page_obj.mediabox
    page_h = float(media_box.height)

    quad_points = []
    all_x1, all_y1_pdf, all_x2, all_y2_pdf = [], [], [], []

    for (_, x1, y1_top, x2, y2_top) in word_span:
        # Flip from top-left to PDF bottom-left
        y1_pdf = page_h - y2_top
        y2_pdf = page_h - y1_top
        quad_points += [x1, y1_pdf, x2, y1_pdf, x1, y2_pdf, x2, y2_pdf]
        all_x1.append(x1); all_x2.append(x2)
        all_y1_pdf.append(y1_pdf); all_y2_pdf.append(y2_pdf)

    rect = [min(all_x1), min(all_y1_pdf), max(all_x2), max(all_y2_pdf)]

    r, g, b = color_rgb
    annot = DictionaryObject({
        NameObject('/Type'):       NameObject('/Annot'),
        NameObject('/Subtype'):    NameObject('/Highlight'),
        NameObject('/Rect'):       ArrayObject([FloatObject(v) for v in rect]),
        NameObject('/QuadPoints'): ArrayObject([FloatObject(v) for v in quad_points]),
        NameObject('/C'):          ArrayObject([FloatObject(r), FloatObject(g), FloatObject(b)]),
        NameObject('/CA'):         FloatObject(0.5),
        NameObject('/F'):          NumberObject(4),
        NameObject('/NM'):         ByteStringObject(
            ''.join(random.choices(string.ascii_letters + string.digits, k=8)).encode()
        ),
    })

    annot_ref = writer._add_object(annot)
    if '/Annots' in page_obj:
        page_obj['/Annots'].append(annot_ref)
    else:
        page_obj[NameObject('/Annots')] = ArrayObject([annot_ref])

    with open(pdf_path, 'wb') as f:
        writer.write(f)


# ---------------------------------------------------------------------------
# Main: run the fallback ladder
# ---------------------------------------------------------------------------

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--pdf',   required=True)
    parser.add_argument('--page',  type=int, required=True)
    parser.add_argument('--quote', required=True)
    parser.add_argument('--color', default='green', choices=COLORS.keys())
    args = parser.parse_args()

    pdf_path = os.path.expanduser(args.pdf)

    tiers = [
        ('pdftotext+xml',  tier1_pdftotext),
        ('pypdf-visitor',  tier3_pypdf),
        ('pdfminer',       tier4_pdfminer),
    ]

    words, W, H = None, None, None
    tier_used = None
    last_errors = []

    for name, fn in tiers:
        try:
            words, W, H = fn(pdf_path, args.page)
            tier_used = name
            print(f"TIER: {name}", file=sys.stderr)
            break
        except Exception as e:
            last_errors.append(f"{name}: {e}")
            print(f"TIER_FAIL: {name}: {e}", file=sys.stderr)

    # Tier 5: visual render - only reached if all text-layer tiers failed
    if words is None:
        try:
            tier5_visual_render(pdf_path, args.page)
        except RuntimeError as e:
            err = str(e)
            if err.startswith('VISUAL_FALLBACK:'):
                img_path = err.split(':', 1)[1]
                # Emit the special signal - citation-hunt will read the image
                # and estimate coords visually, then call write_highlight directly
                print(f"VISUAL_FALLBACK: {img_path}")
                sys.exit(2)  # exit code 2 = visual fallback needed
            last_errors.append(f"visual: {e}")

        print("MATCH_FAILED: all extraction tiers exhausted", file=sys.stderr)
        for e in last_errors:
            print(f"  {e}", file=sys.stderr)
        sys.exit(1)

    # Match the quote
    try:
        span = match_span(words, args.quote)
    except ValueError as e:
        print(f"MATCH_FAILED: {e}", file=sys.stderr)
        print(f"TIER_USED: {tier_used}", file=sys.stderr)
        sys.exit(1)

    # Fractional coords (top-left origin for org-noter)
    x1 = min(w[1] for w in span) / W
    y1 = min(w[2] for w in span) / H
    x2 = max(w[3] for w in span) / W
    y2 = max(w[4] for w in span) / H

    print(f"NOTER_PAGE: ({args.page} {x1:.10f} . {y1:.10f})")
    print(f"HIGHLIGHT: #s(pdf-highlight {args.page} (({y1:.10f} {x1:.10f} {y2:.10f} {x2:.10f})))")

    write_highlight(pdf_path, args.page, span, W, H, COLORS[args.color])
    print(f"WROTE: highlight on page {args.page} of {pdf_path}", file=sys.stderr)


if __name__ == '__main__':
    main()
